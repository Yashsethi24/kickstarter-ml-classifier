.withColumn(
    "MERCHANT_NAME",
    F.when(
        F.col("MERCHANT_NAME").rlike(r"(?i)(amazon|amzn)")
        | (F.upper(F.trim(F.col("MERCHANT_NAME"))) == "BUSINESS PRIME"),
        F.lit("Amazon")
    )
    .when(
        F.col("MERCHANT_NAME").rlike(r"(?i)wal[\s-]*mart"),
        F.lit("Walmart")
    )
    .otherwise(F.col("MERCHANT_NAME"))
)
