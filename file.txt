import pandas as pd
import glob
import os

folder_path = "/Workspace/Users/.../Card_Blockings/FICO"

files = sorted(glob.glob(os.path.join(folder_path, "*.csv")))

df_list = []

for file in files:

    filename = os.path.basename(file)

    # Extract run date
    run_date = pd.to_datetime(
        filename.split("_")[3],
        format="%Y%m%d"
    ).date()

    # Read CSV
    df = pd.read_csv(file, sep="|")

    # Remove extra spaces from column names
    df.columns = df.columns.str.strip()

    # Add run date
    df["Run_Date"] = run_date

    df_list.append(df)

# Check if column structure is consistent
for file, df in zip(files, df_list):
    print(os.path.basename(file), df.shape, df.columns.tolist())

# Concatenate all files
final_df = pd.concat(df_list, ignore_index=True)

display(final_df)
