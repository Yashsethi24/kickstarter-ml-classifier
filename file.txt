from pyspark.sql import functions as F, Window

# One mapping per account and claim month.
claim_months = (
    spark.createDataFrame(final_df[["ACCOUNTNUMBER", "CLM_YM"]])
    .select(
        F.col("ACCOUNTNUMBER").cast("string").alias("CARD_NUMBER"),
        "CLM_YM"
    )
    .filter(F.col("CLM_YM").isNotNull())
    .distinct()
)

merchant_summary = (
    transactions_df
    .join(claim_months, on="CARD_NUMBER", how="inner")
    .withColumn(
        "MERCHANT_NAME",
        F.coalesce(F.trim(F.col("MERCHANT_NAME")), F.lit("UNKNOWN"))
    )
    .groupBy("CLM_YM", "MERCHANT_NAME")
    .agg(
        F.count("*").alias("transaction_count"),
        F.sum("AMOUNT_APPROVED").alias("total_amount_approved")
    )
)

# Each merchant's share of transactions and approved amount that month.
month_window = Window.partitionBy("CLM_YM")

merchant_summary = (
    merchant_summary
    .withColumn(
        "transaction_pct",
        F.round(
            100.0 * F.col("transaction_count")
            / F.sum("transaction_count").over(month_window),
            2
        )
    )
    .withColumn(
        "approved_amount_pct",
        F.round(
            100.0 * F.col("total_amount_approved")
            / F.sum("total_amount_approved").over(month_window),
            2
        )
    )
    .orderBy("CLM_YM", F.desc("total_amount_approved"))
)

display(merchant_summary)
