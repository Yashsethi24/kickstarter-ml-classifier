def om5_timing_nodes(rows):
    delta = rows["OM5_Alert_date"] - rows["OPEN_DT"]

    def timing_node(label, mask):
        subset = rows.loc[mask].copy()

        subset["_om5_status"] = (
            subset["OM5 Action Status"]
            .astype("string")
            .str.strip()
            .replace("", pd.NA)
            .fillna("Missing action status")
        )

        status_children = [
            node(str(status), group)
            for status, group in subset.groupby(
                "_om5_status", sort=True
            )
        ]

        return node(label, subset, status_children)

    children = [
        timing_node(
            "Worked within 1 day",
            delta.between(
                pd.Timedelta(0),
                pd.Timedelta(days=1),
            ),
        ),
        timing_node(
            "Worked >1–2 days",
            delta.gt(pd.Timedelta(days=1))
            & delta.le(pd.Timedelta(days=2)),
        ),
        timing_node(
            "Worked >2–3 days",
            delta.gt(pd.Timedelta(days=2))
            & delta.le(pd.Timedelta(days=3)),
        ),
        timing_node(
            "Worked after 3 days",
            delta.gt(pd.Timedelta(days=3)),
        ),
    ]

    if delta.isna().any():
        children.append(
            timing_node("Timing unavailable", delta.isna())
        )

    if delta.lt(pd.Timedelta(0)).any():
        children.append(
            timing_node(
                "Before open date",
                delta.lt(pd.Timedelta(0)),
            )
        )

    return children
