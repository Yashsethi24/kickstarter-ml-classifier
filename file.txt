# Extract each fact's name and score from the original text
extracted = (
    df["facts_fired"]
    .dropna()
    .astype(str)
    .str.extractall(
        r"(?P<fact>[^|]+?)\s*\[(?:High|Low)\]\s*"
        r"\((?P<score>[+-]?\d+(?:\.\d+)?)\)"
    )
    .reset_index(drop=True)
)

extracted["fact"] = extracted["fact"].str.strip()
extracted["score"] = pd.to_numeric(extracted["score"], errors="raise")

fact_scores = (
    extracted.loc[extracted["fact"].isin(selected_facts)]
    .groupby("fact")["score"]
    .agg(lambda scores: sorted(scores.unique().tolist()))
    .reset_index(name="scores")
)

display(fact_scores)

print(", ".join(
    f"{row.fact}: {row.scores}"
    for row in fact_scores.itertuples(index=False)
))
