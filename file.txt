from datetime import datetime, timedelta

ref_date = datetime.strptime(end_date, "%Y-%m-%d")

last_friday = ref_date - timedelta(days=(ref_date.weekday() - 4) % 7)
last_to_last_friday = last_friday - timedelta(days=7)

week_start = last_to_last_friday.strftime("%Y%m%d")
week_end = last_friday.strftime("%Y%m%d")

print(week_start, week_end)


query = f"""
SELECT
    date_format(to_date(application_date, 'yyyyMMdd'), 'yyyyMM') AS year_mon,
    COUNT(DISTINCT Application_Number) AS num_apps
FROM final_data
WHERE application_date >= '{week_start}'
  AND application_date <= '{week_end}'
GROUP BY date_format(to_date(application_date, 'yyyyMMdd'), 'yyyyMM')
"""
