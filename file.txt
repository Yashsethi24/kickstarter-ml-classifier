import pandas as pd

fraud_cols = ["CONFIRMED_FRAUD_MBNA", "CONFIRMED_FRAUD_TDCT"]
date_cols = ["SEC_FRAUD_STAT_DT_MBNA", "SEC_FRAUD_STAT_DT_TDCT"]

df["CONFIRMED_FRAUD"] = (
    df[fraud_cols]
    .apply(pd.to_numeric, errors="coerce")

)
