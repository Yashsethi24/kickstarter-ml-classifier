from datetime import datetime, timedelta
from dateutil.relativedelta import relativedelta

# Input: Today's date
end_date = datetime.today().date()

# 1. Last Friday
last_friday = end_date - timedelta(days=(end_date.weekday() - 4) % 7)

# 2. Last to last Friday
last_to_last_friday = last_friday - timedelta(days=7)

# 3. This month start date
mtd_start_date = end_date.replace(day=1)

# 4. Last month start date
mtd_1_start_date = (end_date - relativedelta(months=1)).replace(day=1)

# 5. Same date last month (based on last Friday)
mtd_1_end_date = last_friday - relativedelta(months=1)

# 6. First day of the month, 3 months before today
mtd_3_start_date = (
    end_date.replace(day=1) - relativedelta(months=3)
).strftime("%Y%m%d")
# Convert all dates to yyyyMMdd format
last_friday = last_friday.strftime("%Y%m%d")
last_to_last_friday = last_to_last_friday.strftime("%Y%m%d")
mtd_start_date = mtd_start_date.strftime("%Y%m%d")
mtd_1_start_date = mtd_1_start_date.strftime("%Y%m%d")
mtd_1_end_date = mtd_1_end_date.strftime("%Y%m%d")

print(last_friday, last_to_last_friday, mtd_start_date, mtd_1_start_date, mtd_1_end_date)

last_friday = last_friday.strftime("%Y-%m-%d")
last_to_last_friday = last_to_last_friday.strftime("%Y-%m-%d")
mtd_start_date = mtd_start_date.strftime("%Y-%m-%d")
mtd_1_start_date = mtd_1_start_date.strftime("%Y-%m-%d")
mtd_1_end_date = mtd_1_end_date.strftime("%Y-%m-%d")
mtd_3_start_date = mtd_3_start_date.strftime("%Y-%m-%d")
