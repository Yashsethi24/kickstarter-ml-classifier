# Check whether repeated application rows have different balances.
balance_counts = (
    df.groupby("APPLICATION_NUMBER")["total_current_balance"]
      .nunique(dropna=False)
)

if balance_counts.gt(1).any():
    raise ValueError(
        "Some applications have different balances across rows. "
        "Determine whether these are separate cards or historical snapshots "
        "before calculating an application-level balance."
    )

# One row per application:
# - Union all fired facts across its rows.
# - Mark fraud if any row has fraud = 1.
# - Keep its repeated application balance once.
df = (
    df.groupby("APPLICATION_NUMBER", as_index=False)
      .agg(
          fraud=("fraud", "max"),
          positive_balance=("positive_balance", "first"),
          fact_set=("fact_set", lambda values: set().union(*values)),
      )
)
