from pyspark.sql import functions as F

binary_features = (
    long_df
    .select(ID, "fact")
    .distinct()
    .withColumn("fired", F.lit(1))
    .groupBy(ID)
    .pivot("fact")
    .agg(F.max("fired"))
)

feature_columns = [c for c in binary_features.columns if c != ID]

# Assumes one row per application in model_df.
binary_df = (
    model_df.select(ID, TARGET)
    .join(binary_features, on=ID, how="left")
    .fillna(0, subset=feature_columns)
)

display(binary_df)


import numpy as np
import pandas as pd

from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import average_precision_score, roc_auc_score

binary_pdf = binary_df.toPandas()

X_binary = binary_pdf[feature_columns].astype("int8")
y_binary = binary_pdf[TARGET].astype(int)

X_train_b, X_test_b, y_train_b, y_test_b = train_test_split(
    X_binary,
    y_binary,
    test_size=0.20,
    stratify=y_binary,
    random_state=42
)

binary_model = LogisticRegression(
    max_iter=2000,
    random_state=42
)
binary_model.fit(X_train_b, y_train_b)

test_probability = binary_model.predict_proba(X_test_b)[:, 1]

print("Test average precision:",
      average_precision_score(y_test_b, test_probability))
print("Test ROC AUC:",
      roc_auc_score(y_test_b, test_probability))


binary_weights = pd.DataFrame({
    "fact": X_binary.columns,
    "weight": binary_model.coef_[0]
})

# Your requested integer version: multiply by 100, truncate decimals.
binary_weights["weight_int"] = np.trunc(
    binary_weights["weight"] * 100
).astype("int64")

binary_weights = binary_weights.sort_values(
    "weight", ascending=False
)

print("Intercept:", binary_model.intercept_[0])
display(binary_weights)

binary_application_scores = binary_pdf[[ID, TARGET]].copy()

# Logistic regression score = intercept + sum of fired fact weights.
binary_application_scores["model_score"] = (
    binary_model.decision_function(X_binary)
)

# Estimated fraud probability expressed on a 0–100 scale.
binary_application_scores["risk_score_100"] = (
    binary_model.predict_proba(X_binary)[:, 1] * 100
)

# Your custom integer score = sum of integer weights of fired facts.
integer_weights = np.trunc(
    binary_model.coef_[0] * 100
).astype("int64")

binary_application_scores["total_integer_score"] = (
    X_binary.to_numpy(dtype="int64") @ integer_weights
)

display(binary_application_scores)
