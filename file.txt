mbna_8day_report = f"""
WITH parameters AS (
    SELECT
        DATE_SUB(CURRENT_DATE(), 1) AS report_date,
        DATE_SUB(CURRENT_DATE(), 8) AS report_start,
        DATE_SUB(CURRENT_DATE(), 98) AS benchmark_start,
        DATE_SUB(CURRENT_DATE(), 9) AS benchmark_end,
        8 AS report_days,
        90 AS benchmark_days
),

base_data AS (
    SELECT
        TO_DATE(Application_Date) AS application_date,

        CASE
            WHEN UPPER(TRIM(Address_Province))
                IN ('ON', 'ONTARIO') THEN 'ON'

            WHEN UPPER(TRIM(Address_Province))
                IN ('QC', 'PQ', 'QUEBEC', 'QUÉBEC') THEN 'QC'

            WHEN UPPER(TRIM(Address_Province))
                IN ('AB', 'ALBERTA') THEN 'AB'

            WHEN UPPER(TRIM(Address_Province))
                IN ('BC', 'BRITISH COLUMBIA') THEN 'BC'

            ELSE 'Others'
        END AS province_group,

        CASE
            WHEN Transit_Number = 'AMAZXX' THEN 'AMAZON'
            ELSE 'MBNA'
        END AS channel,

        CAST(Application_Number AS STRING)
            AS application_number,

        COALESCE(
            TRY_CAST(
                REGEXP_REPLACE(
                    CAST(Credit_Limit AS STRING), ',', ''
                ) AS DECIMAL(18, 2)
            ), 0
        ) AS credit_limit

    FROM hive_metastore.fiqm.applications

    WHERE Product_Type = 'PUCC'
        AND DATA_SOURCE = 'MBNA'
        AND Application_Number IS NOT NULL
        AND Application_Number RLIKE '^[A-E]'
        AND TO_DATE(Application_Date) BETWEEN
            DATE_SUB(CURRENT_DATE(), 98)
            AND DATE_SUB(CURRENT_DATE(), 1)
),

-- Aggregate claims data to avoid duplicates

claims_data AS (
    SELECT
        CAST(Application_Number AS STRING)
            AS application_number,

        MAX(
            CASE
                WHEN priority_alert_ind = 1 THEN 1
                ELSE 0
            END
        ) AS priority_alert_ind,

        MAX(
            CASE
                WHEN CONF_APP_FRAUD = 1 OR claim_ind = 1
                    THEN 1
                ELSE 0
            END
        ) AS fraud_ind

    FROM claims_final_ss11

    GROUP BY CAST(Application_Number AS STRING)
),

om5_data AS (
    SELECT DISTINCT
        CAST(Application_Number AS STRING)
            AS application_number

    FROM om5_alerts
),

-- Add alert and fraud indicators

base_w_alerts AS (
    SELECT
        a.*,

        COALESCE(b.priority_alert_ind, 0)
            AS priority_alert_ind,

        CASE
            WHEN c.application_number IS NOT NULL
                THEN 1
            ELSE 0
        END AS OM5_Alert,

        COALESCE(b.fraud_ind, 0) AS fraud_ind

    FROM base_data a

    LEFT JOIN claims_data b
        ON a.application_number = b.application_number

    LEFT JOIN om5_data c
        ON a.application_number = c.application_number
),

channel_data AS (
    SELECT
        *,

        CASE
            WHEN OM5_Alert = 1 OR priority_alert_ind = 1
                THEN 1
            ELSE 0
        END AS alert_ind

    FROM base_w_alerts

    WHERE channel IN ('MBNA', 'AMAZON')
),

-- Previous 90-day historical benchmark

benchmark AS (
    SELECT
        b.channel,
        b.province_group,

        COUNT(DISTINCT b.application_number)
            AS benchmark_application_count,

        SUM(b.credit_limit)
            AS benchmark_credit_limit,

        COUNT(DISTINCT CASE
            WHEN b.alert_ind = 1
            THEN b.application_number
        END) AS benchmark_alerts_count,

        COUNT(DISTINCT CASE
            WHEN b.fraud_ind = 1
            THEN b.application_number
        END) AS benchmark_frauds_count

    FROM channel_data b
    CROSS JOIN parameters p

    WHERE b.application_date BETWEEN
        p.benchmark_start AND p.benchmark_end

    GROUP BY
        b.channel,
        b.province_group
),

-- Actual metrics for the last 8 days

last_8days_actual AS (
    SELECT
        b.channel,
        b.province_group,

        COUNT(DISTINCT b.application_number)
            AS actual_application_count,

        SUM(b.credit_limit)
            AS actual_credit_limit,

        COUNT(DISTINCT CASE
            WHEN b.alert_ind = 1
            THEN b.application_number
        END) AS actual_alerts_count,

        COUNT(DISTINCT CASE
            WHEN b.fraud_ind = 1
            THEN b.application_number
        END) AS actual_frauds_count

    FROM channel_data b
    CROSS JOIN parameters p

    WHERE b.application_date BETWEEN
        p.report_start AND p.report_date

    GROUP BY
        b.channel,
        b.province_group
),

channel_list AS (
    SELECT channel
    FROM VALUES
        ('MBNA'),
        ('AMAZON')
    AS channel_list(channel)
),

province_list AS (
    SELECT province_group
    FROM VALUES
        ('ON'),
        ('QC'),
        ('AB'),
        ('BC'),
        ('Others')
    AS province_list(province_group)
),

report_groups AS (
    SELECT
        c.channel,
        g.province_group

    FROM channel_list c
    CROSS JOIN province_list g
),

report_metrics AS (
    SELECT
        p.report_date,
        p.report_start,
        p.benchmark_start,
        p.benchmark_end,

        rg.channel,
        rg.province_group,

        -- APPLICATION COUNT

        CAST(
            COALESCE(a.actual_application_count, 0)
        AS INT) AS actual_application_count,

        CAST(ROUND(
            COALESCE(b.benchmark_application_count, 0)
            / NULLIF(p.benchmark_days, 0)
            * p.report_days, 0
        ) AS INT) AS expected_application_count,

        -- CREDIT LIMIT

        CONCAT(
            '$',
            FORMAT_NUMBER(
                COALESCE(a.actual_credit_limit, 0), 0
            )
        ) AS actual_credit_limit,

        CONCAT(
            '$',
            FORMAT_NUMBER(
                COALESCE(b.benchmark_credit_limit, 0)
                / NULLIF(p.benchmark_days, 0)
                * p.report_days, 0
            )
        ) AS expected_credit_limit,

        -- ALERT COUNT

        CAST(
            COALESCE(a.actual_alerts_count, 0)
        AS INT) AS actual_alerts_count,

        CAST(ROUND(
            COALESCE(b.benchmark_alerts_count, 0)
            / NULLIF(p.benchmark_days, 0)
            * p.report_days, 0
        ) AS INT) AS expected_alerts_count,

        -- FRAUD COUNT

        CAST(
            COALESCE(a.actual_frauds_count, 0)
        AS INT) AS actual_frauds_count,

        CAST(ROUND(
            COALESCE(b.benchmark_frauds_count, 0)
            / NULLIF(p.benchmark_days, 0)
            * p.report_days, 0
        ) AS INT) AS expected_frauds_count

    FROM report_groups rg
    CROSS JOIN parameters p

    LEFT JOIN benchmark b
        ON rg.channel = b.channel
        AND rg.province_group = b.province_group

    LEFT JOIN last_8days_actual a
        ON rg.channel = a.channel
        AND rg.province_group = a.province_group
)

SELECT
    *,

    -- APPLICATION THRESHOLD

    CASE
        WHEN expected_application_count = 0
            AND actual_application_count = 0
            THEN '🟢Green'

        WHEN expected_application_count = 0
            AND actual_application_count > 0
            THEN '🔴Red'

        WHEN actual_application_count
            <= expected_application_count * 1.25
            THEN '🟢Green'

        WHEN actual_application_count
            <= expected_application_count * 1.50
            THEN '🟠Yellow'

        ELSE '🔴Red'
    END AS Threshold,

    -- ALERT THRESHOLD

    CASE
        WHEN expected_alerts_count = 0
            AND actual_alerts_count = 0
            THEN '🟢Green'

        WHEN expected_alerts_count = 0
            AND actual_alerts_count > 0
            THEN '🔴Red'

        WHEN actual_alerts_count
            <= expected_alerts_count * 1.25
            THEN '🟢Green'

        WHEN actual_alerts_count
            <= expected_alerts_count * 1.50
            THEN '🟠Yellow'

        ELSE '🔴Red'
    END AS Alerts_Threshold,

    -- FRAUD THRESHOLD

    CASE
        WHEN expected_frauds_count = 0
            AND actual_frauds_count = 0
            THEN '🟢Green'

        WHEN expected_frauds_count = 0
            AND actual_frauds_count > 0
            THEN '🔴Red'

        WHEN actual_frauds_count
            <= expected_frauds_count * 1.25
            THEN '🟢Green'

        WHEN actual_frauds_count
            <= expected_frauds_count * 1.50
            THEN '🟠Yellow'

        ELSE '🔴Red'
    END AS Frauds_Threshold

FROM report_metrics

ORDER BY
    CASE channel
        WHEN 'MBNA' THEN 1
        WHEN 'AMAZON' THEN 2
        ELSE 3
    END,

    CASE province_group
        WHEN 'ON' THEN 1
        WHEN 'QC' THEN 2
        WHEN 'AB' THEN 3
        WHEN 'BC' THEN 4
        ELSE 5
    END
"""
