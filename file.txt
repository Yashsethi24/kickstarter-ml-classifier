import re
import pandas as pd
from sklearn.preprocessing import MultiLabelBinarizer

def parse_facts(value):
    if pd.isna(value) or not str(value).strip():
        return []

    facts = []
    for item in str(value).split("|"):
        # Remove trailing current weight, including negative weights.
        fact = re.sub(r"\s*\(-?\d+(?:\.\d+)?\)\s*$", "", item.strip())

        # Remove trailing risk annotation.
        fact = re.sub(
            r"\s*\[(?:High|Medium|Low)\]\s*$",
            "",
            fact,
            flags=re.IGNORECASE,
        ).strip()

        if fact:
            facts.append(fact)

    return sorted(set(facts))
