import numpy as np
import pandas as pd

from sklearn.preprocessing import MultiLabelBinarizer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import average_precision_score, roc_auc_score

# --------------------------------------------------
# 1. Validate and split chronologically
# --------------------------------------------------
data = df.copy()
data["LOAD_DATE"] = pd.to_datetime(data["LOAD_DATE"], errors="raise")
data["fraud"] = pd.to_numeric(data["fraud"], errors="raise")

if data["LOAD_DATE"].isna().any():
    raise ValueError("LOAD_DATE contains missing values.")

if not data["fraud"].isin([0, 1]).all():
    raise ValueError("fraud must contain only 0 and 1.")

data = data.sort_values("LOAD_DATE").reset_index(drop=True)

# Keep every application from the same calendar day in the same split.
data["_day"] = data["LOAD_DATE"].dt.normalize()
dates = np.sort(data["_day"].unique())

if len(dates) < 5:
    raise ValueError("Need at least five distinct application dates.")

train_end = dates[int(len(dates) * 0.60)]
validation_end = dates[int(len(dates) * 0.80)]

train = data[data["_day"] < train_end].copy()
validation = data[
    (data["_day"] >= train_end) &
    (data["_day"] < validation_end)
].copy()
test = data[data["_day"] >= validation_end].copy()

for name, part in [
    ("Train", train),
    ("Validation", validation),
    ("Test", test),
]:
    if part["fraud"].nunique() != 2:
        raise ValueError(f"{name} must contain fraud and legitimate cases.")

    print(
        f"{name}: {len(part):,} applications, "
        f"fraud rate = {part['fraud'].mean():.2%}, "
        f"dates = {part['_day'].min().date()} to "
        f"{part['_day'].max().date()}"
    )

# --------------------------------------------------
# 2. Learn fact vocabulary using training data only
# --------------------------------------------------
encoder = MultiLabelBinarizer(sparse_output=True)
X_train = encoder.fit_transform(train["fact_list"])

if X_train.shape[1] == 0:
    raise ValueError("No facts found in training data.")

known_facts = set(encoder.classes_)

def encode_facts(part):
    # New facts cannot receive a learned weight until retraining.
    return encoder.transform(
        part["fact_list"].apply(
            lambda facts: [f for f in facts if f in known_facts]
        )
    )

X_validation = encode_facts(validation)
X_test = encode_facts(test)

y_train = train["fraud"].astype(int).to_numpy()
y_validation = validation["fraud"].astype(int).to_numpy()
y_test = test["fraud"].astype(int).to_numpy()

# --------------------------------------------------
# 3. Select regularization using validation performance
# --------------------------------------------------
candidates = []

for C in [0.01, 0.1, 1.0, 10.0]:
    candidate = LogisticRegression(
        C=C,
        penalty="l2",
        solver="lbfgs",
        max_iter=3000,
    )
    candidate.fit(X_train, y_train)

    probability = candidate.predict_proba(X_validation)[:, 1]
    ap = average_precision_score(y_validation, probability)

    candidates.append((ap, C, candidate))

best_ap, best_C, model = max(candidates, key=lambda item: item[0])

print(f"\nSelected C: {best_C}")
print(f"Validation average precision: {best_ap:.4f}")

# --------------------------------------------------
# 4. Export learned weights and training evidence
# --------------------------------------------------
fact_counts = np.asarray(X_train.sum(axis=0)).ravel()
fraud_counts = np.asarray(X_train.T @ y_train).ravel()

weights = pd.DataFrame({
    "fact": encoder.classes_,
    "learned_weight": model.coef_[0],
    "training_applications": fact_counts.astype(int),
    "training_fraud_applications": fraud_counts.astype(int),
    "training_fraud_rate": fraud_counts / fact_counts,
}).sort_values("learned_weight", ascending=False)

print(f"\nModel intercept: {model.intercept_[0]:.6f}")
print(weights.head(20).to_string(index=False))

weights.to_csv("proposed_fact_weights.csv", index=False)

# These weights are log-odds coefficients, not your current points scale.
# Keep full precision when scoring.

# --------------------------------------------------
# 5. Set thresholds using validation data only
# --------------------------------------------------
TARGET_DECLINE_RATE = 0.05
TARGET_REVIEW_RATE = 0.10

if not (
    0 < TARGET_DECLINE_RATE < 1
    and 0 < TARGET_REVIEW_RATE
    and TARGET_DECLINE_RATE + TARGET_REVIEW_RATE < 1
):
    raise ValueError("Review and decline rates must sum to less than 1.")

validation_probability = model.predict_proba(X_validation)[:, 1]

decline_threshold = np.quantile(
    validation_probability,
    1 - TARGET_DECLINE_RATE,
)
review_threshold = np.quantile(
    validation_probability,
    1 - TARGET_DECLINE_RATE - TARGET_REVIEW_RATE,
)

print(f"\nReview probability threshold: {review_threshold:.6f}")
print(f"Decline probability threshold: {decline_threshold:.6f}")

# --------------------------------------------------
# 6. Evaluate once on the later test period
# --------------------------------------------------
test_probability = model.predict_proba(X_test)[:, 1]

test["predicted_fraud_probability"] = test_probability
test["proposed_decision"] = np.select(
    [
        test_probability >= decline_threshold,
        test_probability >= review_threshold,
    ],
    ["decline", "review"],
    default="approve",
)

summary = (
    test.groupby("proposed_decision")
    .agg(
        applications=("fraud", "size"),
        fraud_applications=("fraud", "sum"),
    )
    .reindex(["approve", "review", "decline"], fill_value=0)
)

summary["legitimate_applications"] = (
    summary["applications"] - summary["fraud_applications"]
)
summary["application_share"] = (
    summary["applications"] / len(test)
)
summary["fraud_rate"] = (
    summary["fraud_applications"]
    / summary["applications"].replace(0, np.nan)
)

total_fraud = int(y_test.sum())
total_legitimate = int((y_test == 0).sum())

fraud_declined = summary.loc["decline", "fraud_applications"]
fraud_reviewed = summary.loc["review", "fraud_applications"]
legitimate_declined = summary.loc["decline", "legitimate_applications"]

print("\nTest results:")
print(summary.to_string())

print(f"\nROC AUC: {roc_auc_score(y_test, test_probability):.4f}")
print(
    "Average precision: "
    f"{average_precision_score(y_test, test_probability):.4f}"
)
print(f"Fraud routed to decline: {fraud_declined / total_fraud:.2%}")
print(
    "Fraud routed to review or decline: "
    f"{(fraud_declined + fraud_reviewed) / total_fraud:.2%}"
)
print(
    "Legitimate applications declined: "
    f"{legitimate_declined / total_legitimate:.2%}"
)

test.to_csv("test_applications_with_proposed_scores.csv", index=False)
summary.to_csv("test_decision_summary.csv")
