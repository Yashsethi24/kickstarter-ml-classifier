import pandas as pd
import networkx as nx

def remove_cross_duplicates(df):

    id_col = 'Unique_record_ID'
    app_col = 'Application_Number'

    # Remove duplicate ID-Application combinations
    df = df.drop_duplicates(
        subset=[id_col, app_col]
    ).copy()

    # Count occurrences
    id_count = df.groupby(id_col)[app_col].transform('size')
    app_count = df.groupby(app_col)[id_col].transform('size')

    # Keep records that have no duplicates on either side
    unique_df = df[(id_count == 1) & (app_count == 1)]

    # Remaining records
    duplicate_df = df[~((id_count == 1) & (app_count == 1))]

    # Create graph
    G = nx.Graph()

    for uid, app in duplicate_df[[id_col, app_col]].itertuples(
        index=False, name=None
    ):
        G.add_edge(('ID', uid), ('APP', app))

    # Maximum matching
    matching = nx.algorithms.bipartite.maximum_matching(
        G,
        top_nodes={
            ('ID', uid)
            for uid in duplicate_df[id_col].unique()
        }
    )

    matched_pairs = [
        (uid[1], app[1])
        for uid, app in matching.items()
        if uid[0] == 'ID'
    ]

    matched_df = pd.DataFrame(
        matched_pairs,
        columns=[id_col, app_col]
    )

    matched_records = duplicate_df.merge(
        matched_df,
        on=[id_col, app_col],
        how='inner'
    )

    # Combine unique records and resolved duplicates
    final_df = pd.concat(
        [unique_df, matched_records],
        ignore_index=True
    )

    return final_df
