# 1. Card-age distribution: loss value and application count
age_columns = [
    "<1 month", "1–3 months", ">3 months", "Unknown / invalid"
]

age_summary = (
    df.groupby(["claim_month", "card_age"])
      .agg(
          Loss_value=("tot_loss", "sum"),
          Applications=("Application_Number", "nunique"),
      )
      .unstack("card_age")
      .reindex(
          index=months,
          columns=pd.MultiIndex.from_product([
              ["Loss_value", "Applications"], age_columns
          ]),
      )
      .fillna(0)
)

# Put value and count beside each other for each age bucket.
age_summary = age_summary.swaplevel(0, 1, axis=1).reindex(
    columns=pd.MultiIndex.from_product([
        age_columns, ["Loss_value", "Applications"]
    ])
)

age_summary.columns = [
    f"{bucket} | {metric}"
    for bucket, metric in age_summary.columns
]

# 2. Alert distribution: loss value and application count
alert_summary = pd.DataFrame(index=months)

groups = {
    "All applications": pd.Series(True, index=df.index),
    "Priority alert": pd.to_numeric(
        df["priority_alert_ind"], errors="coerce"
    ).eq(1),
    "Auto block": pd.to_numeric(
        df["autoblock_alert_ind"], errors="coerce"
    ).eq(1),
    "Address mismatch": pd.to_numeric(
        df["add_mismatch_ind"], errors="coerce"
    ).eq(1),
}

for label, mask in groups.items():
    summary = (
        df.loc[mask]
          .groupby("claim_month")
          .agg(
              Loss_value=("tot_loss", "sum"),
              Applications=("Application_Number", "nunique"),
          )
          .reindex(months, fill_value=0)
    )

    alert_summary[f"{label} | Loss_value"] = summary["Loss_value"]
    alert_summary[f"{label} | Applications"] = summary["Applications"]

# Format both outputs.
for result in [age_summary, alert_summary]:
    count_cols = [
        col for col in result.columns
        if col.endswith("| Applications")
    ]
    result[count_cols] = result[count_cols].astype(int)
    result.index = result.index.strftime("%Y-%m")
    result.index.name = "Claim month"

age_summary = age_summary.reset_index()
alert_summary = alert_summary.reset_index()

display(age_summary)
display(alert_summary)
