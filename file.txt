# Month based on claim date, not transaction date.
claims_df["CLAIM_YEAR_MONTH"] = (
    claims_df["FST_CLM_DT"].dt.strftime("%Y-%m")
)

same_month_claims_distribution = (
    claims_df.loc[claims_df["CLAIM_TIMING"].eq("Same Month")]
    .groupby("CLAIM_YEAR_MONTH")
    .agg(number_of_claims=("PYID", "nunique"))
    .reset_index()
    .sort_values("CLAIM_YEAR_MONTH")
)

display(same_month_claims_distribution)
