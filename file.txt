APP = "Application_Number"

strategy_cols = [
    "priority_alert_ind",
    "om5_alert_ind",
    "add_mismatch_ind",
]

# Handles flags stored as either numeric 1 or string "1".
claims_df["ANY_STRATEGY_ALERTED"] = (
    claims_df[strategy_cols]
    .apply(pd.to_numeric, errors="coerce")
    .eq(1)
    .any(axis=1)
    .astype(int)
)

# Collapse repeated transaction/card rows to application level.
application_level = (
    claims_df.groupby(["CLAIM_YM", APP], as_index=False)
    .agg(ANY_STRATEGY_ALERTED=("ANY_STRATEGY_ALERTED", "max"))
)

# Final monthly view.
application_distribution = (
    application_level.groupby("CLAIM_YM", as_index=False)
    .agg(
        unique_applications=(APP, "nunique"),
        alerted_applications=("ANY_STRATEGY_ALERTED", "sum"),
    )
)

application_distribution["not_alerted_applications"] = (
    application_distribution["unique_applications"]
    - application_distribution["alerted_applications"]
)

application_distribution["alerted_pct"] = (
    application_distribution["alerted_applications"]
    / application_distribution["unique_applications"]
    * 100
).round(2)

display(application_distribution)
