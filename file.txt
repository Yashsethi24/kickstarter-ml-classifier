# 2. Prepare data.
df = final_df[
    [
        "APPLICATION_NUMBER",
        "LOAD_DATE",
        "fraud",
        "facts_fired",
        "total_current_balance",
    ]
].copy()

df["LOAD_DATE"] = pd.to_datetime(df["LOAD_DATE"], errors="raise")

if df[["APPLICATION_NUMBER", "LOAD_DATE"]].isna().any().any():
    raise ValueError("APPLICATION_NUMBER or LOAD_DATE contains missing values.")

df["application_month"] = df["LOAD_DATE"].dt.to_period("M").astype(str)

df["fraud"] = pd.to_numeric(
    df["fraud"], errors="raise"
).fillna(0)

if not df["fraud"].isin([0, 1]).all():
    raise ValueError("fraud must contain only 0 or 1.")

df["total_current_balance"] = pd.to_numeric(
    df["total_current_balance"], errors="raise"
)

# Confirm each application belongs to one month.
month_counts = (
    df.groupby("APPLICATION_NUMBER")["application_month"].nunique()
)

if month_counts.gt(1).any():
    raise ValueError(
        "Some applications have LOAD_DATE values in different months. "
        "Choose the intended application load date before continuing."
    )

# Confirm repeated rows have the same application-level balance.
balance_counts = (
    df.groupby("APPLICATION_NUMBER")["total_current_balance"]
      .nunique(dropna=False)
)

if balance_counts.gt(1).any():
    raise ValueError(
        "Some applications have different balances across rows. "
        "Resolve card-level balances or historical snapshots first."
    )

df["positive_balance"] = (
    df["total_current_balance"].clip(lower=0).fillna(0)
)

def clean_fact(value):
    value = str(value).strip()
    value = re.sub(r"\s*\([-+]?\d+(?:\.\d+)?\)\s*$", "", value)
    value = re.sub(r"\s*\[[^\]]*\]\s*$", "", value)
    return value.strip()

df["fact_set"] = df["facts_fired"].fillna("").map(
    lambda value: {
        clean_fact(part)
        for part in str(value).split("|")
        if part.strip()
    }
)

# One row per application, combining all fired facts.
df = (
    df.groupby("APPLICATION_NUMBER", as_index=False)
      .agg(
          application_month=("application_month", "first"),
          fraud=("fraud", "max"),
          positive_balance=("positive_balance", "first"),
          fact_set=("fact_set", lambda values: set().union(*values)),
      )
)

# 3. Calculate incremental performance independently for each month.
results = []

for month, monthly_df in df.groupby("application_month", sort=True):
    remaining = pd.Series(True, index=monthly_df.index)

    for position, row in enumerate(
        selected.itertuples(index=False), start=1
    ):
        fact_name = clean_fact(row.fact)

        matched = remaining & monthly_df["fact_set"].map(
            lambda facts: fact_name in facts
        )

        captured = monthly_df.loc[matched]
        fraud_mask = captured["fraud"].eq(1)

        applications = len(captured)
        fraud_applications = int(fraud_mask.sum())
        nonfraud_applications = applications - fraud_applications

        results.append({
            "application_month": month,
            "order": position,
            "fact": row.fact,
            "original_FPR": row.FPR,
            "incremental_applications": applications,
            "incremental_fraud_applications": fraud_applications,
            "incremental_nonfraud_applications": nonfraud_applications,
            "incremental_FPR": (
                nonfraud_applications / fraud_applications
                if fraud_applications else np.nan
            ),
            "incremental_fraud_balance": captured.loc[
                fraud_mask, "positive_balance"
            ].sum(),
        })

        remaining.loc[matched] = False

monthly_incremental_performance = pd.DataFrame(results)
