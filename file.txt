import pandas as pd

# Clean status values
secure_fraud = slim_pd["STC_SECURE_FRAUD"].astype("string").str.strip()
tiered_auth = slim_pd["STATC_TIERED_AUTH"].astype("string").str.strip()

# Convert dates — set format explicitly if needed, e.g. format="%Y%m%d"
open_date = pd.to_datetime(
    slim_pd["OPEN_DT"].astype("string").str.strip(),
    errors="coerce",
)

fraud_date = pd.to_datetime(
    slim_pd["SEC_FRAUD_STAT_DT"].astype("string").str.strip(),
    errors="coerce",
)

slim_pd["days_to_fraud"] = (
    fraud_date.dt.normalize() - open_date.dt.normalize()
).dt.days

status_present = (
    secure_fraud.notna()
    & ~secure_fraud.str.lower().isin(["", "null", "none", "nan"])
)

slim_pd["CONFIRMED_FRAUD"] = (
    status_present
    & slim_pd["days_to_fraud"].between(0, 90)
).fillna(False).astype(int)

slim_pd["SUSPECTED_FRAUD"] = (
    tiered_auth.str.upper().eq("L7")
).fillna(False).astype(int)
