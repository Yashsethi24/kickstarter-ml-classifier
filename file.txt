    .astype("string")
    .str.strip()
    .dropna()
)

combinations_to_check = combinations_to_check[
    ~combinations_to_check.str.lower().isin(["", "nan", "none", "null"])

values = present.fillna(False).to_numpy(dtype=bool)
column_positions = {
    name: position for position, name in enumerate(present.columns)
}

# Parse and validate once
rules = []
for combination in combinations_to_check:
    facts = [fact.strip() for fact in combination.split(" + ")]

    missing = [fact for fact in facts if fact not in column_positions]
    if missing:
        raise ValueError(f"Facts missing from present: {missing}")

    rules.append([column_positions[fact] for fact in facts])

# Match each application once, regardless of overlapping combinations
matched = np.zeros(len(present), dtype=bool)

for positions in rules:
    rule_match = values[:, positions[0]].copy()

    for position in positions[1:]:
        rule_match &= values[:, position]

    matched |= rule_match

    if matched.all():
        break

print("Combinations selected:", len(rules))
print("Rows checked:", len(matched))
print("Rows matching combinations:", matched.sum())
