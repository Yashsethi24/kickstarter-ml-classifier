# ------------------------------------------------------------
# 1. Monthly alerts summary
# Count each application once within each run month.
# ------------------------------------------------------------
base["run_month"] = base["RUN_DATE"].dt.to_period("M").astype("string")

monthly_apps = (
    base.dropna(subset=["APPLICATION_NUMBER", "RUN_DATE"])
    .groupby(["run_month", "APPLICATION_NUMBER"], as_index=False)
    .agg(fraud_status=("CNF_FRAUD_IND", "max"))
)

monthly_apps["genuine"] = monthly_apps["fraud_status"].eq(0)
monthly_apps["fraud"] = monthly_apps["fraud_status"].eq(1)
monthly_apps["unknown"] = monthly_apps["fraud_status"].isna()

monthly_alerts_summary = (
    monthly_apps.groupby("run_month", as_index=False)
    .agg(
        alerts=("APPLICATION_NUMBER", "size"),
        genuine_applications=("genuine", "sum"),
        frauds=("fraud", "sum"),
        unknown_status=("unknown", "sum")
    )
)

monthly_alerts_summary["FPR"] = (
    monthly_alerts_summary["genuine_applications"]
    .div(monthly_alerts_summary["frauds"].replace(0, np.nan))
)

display(monthly_alerts_summary)


# ------------------------------------------------------------
# 2. Monthly utilization summary — unblocked cards
# Uses the card-level report from the previous code.
# ------------------------------------------------------------
monthly_unblocked = report.loc[report["unblocked"]].copy()

monthly_unblocked["run_month"] = (
    monthly_unblocked["RUN_DATE"].dt.to_period("M").astype("string")
)

bands = ["0–10%", "10–30%", "30–50%", "50–80%", "80%+",
         "Unknown credit limit"]

monthly_unblocked["utilization_band"] = (
    pd.cut(
        monthly_unblocked["utilization_pct"],
        bins=[0, 10, 30, 50, 80, np.inf],
        labels=bands[:-1],
        right=False
    )
    .astype("string")
    .fillna("Unknown credit limit")
)

monthly_utilization_summary = (
    monthly_unblocked
    .groupby(["run_month", "utilization_band"])["CARD_NUMBER"]
    .nunique()
    .unstack("utilization_band", fill_value=0)
    .reindex(columns=bands, fill_value=0)
    .rename_axis(columns=None)
    .reset_index()
)

monthly_utilization_summary["total_unblocked_cards"] = (
    monthly_utilization_summary[bands].sum(axis=1)
)

display(monthly_utilization_summary)
