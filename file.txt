application_numbers = (
    apps["APPLICATION_NUMBER"]
    .dropna()
    .astype(str)
    .str.strip()
)
application_numbers = application_numbers[
    application_numbers != ""
].drop_duplicates()

ids_df = spark.createDataFrame(
    [(app_id,) for app_id in application_numbers],
    schema="application_number STRING"
)

ids_df.createOrReplaceTempView("selected_application_ids")


%sql
SELECT * FROM selected_application_ids;
