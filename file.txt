SELECT *,
    LOWER(TRIM(CONCAT_WS(' ',
        CASE
            WHEN UPPER(TRIM(PRIMARY_FIRST_NAME)) = 'NULL'
            THEN NULL
            ELSE PRIMARY_FIRST_NAME
        END,
        CASE
            WHEN UPPER(TRIM(PRIMARY_FAMILY_NAME)) = 'NULL'
            THEN NULL
            ELSE PRIMARY_FAMILY_NAME
        END
    ))) AS l_name
FROM hive_metastore.fiqm.applications)))
