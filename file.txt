from pathlib import Path
import pandas as pd

BASE_PATH = Path(
    "/Workspace/Users/rajshreyas.penukonda@td.com/"
    "Applications Fraud/TU IDX/Backbook"
)

SCORED_PATH = BASE_PATH / "Scored_Files" / "8681"

files = [
    ("TDCT", "genuine", "Genuines", "20261015_TDCT_Good_Accts.txt"),
    ("TDCT", "fraud",   "Frauds",   "20261015_TDCT_Frauds.txt"),
    ("TDCT", "DLQ",     "DLQ",      "20261015_TDCT_DLQ.txt"),
    ("MBNA", "genuine", "Genuines", "20261015_MBNA_Good_accts.txt"),
    ("MBNA", "fraud",   "Frauds",   "20261015_MBNA_Frauds.txt"),
    ("MBNA", "DLQ",     "DLQ",      "20261015_MBNA_DLQ.txt"),
]

dataframes = []
summary = []

for portfolio, category, mapping_folder, filename in files:
    main_path = SCORED_PATH / filename
    mapping_path = (
        BASE_PATH
        / portfolio
        / mapping_folder
        / "application_id_mapping.csv"
    )

    main_df = pd.read_csv(
        main_path,
        sep="|",
        dtype=str,
        encoding="utf-8-sig",
    )

    mapping_df = pd.read_csv(
        mapping_path,
        sep=",",
        dtype=str,
        encoding="utf-8-sig",
    )

    main_df.columns = main_df.columns.str.strip()
    mapping_df.columns = mapping_df.columns.str.strip()

    if len(main_df) != len(mapping_df):
        raise ValueError(
            f"{portfolio}/{category}: row counts differ. "
            f"Scored file={len(main_df):,}; "
            f"mapping file={len(mapping_df):,}."
        )

    # Preserve every row and its original position.
    main_df = main_df.reset_index(drop=True)
    mapping_df = mapping_df.reset_index(drop=True)

    # Suffix overlapping mapping columns to keep both versions.
    mapping_df = mapping_df.rename(columns={
        column: f"{column}_mapping"
        for column in mapping_df.columns
        if column in main_df.columns
    })

    combined_df = pd.concat([main_df, mapping_df], axis=1)

    if not combined_df.columns.is_unique:
        raise ValueError(
            f"{portfolio}/{category}: duplicate column names "
            "remain after adding the _mapping suffix."
        )

    combined_df["category"] = category
    combined_df["portfolio"] = portfolio

    dataframes.append(combined_df)

    summary.append({
        "portfolio": portfolio,
        "category": category,
        "scored_rows": len(main_df),
        "mapping_rows": len(mapping_df),
        "combined_rows": len(combined_df),
    })

final_df = pd.concat(dataframes, ignore_index=True, sort=False)

display(pd.DataFrame(summary))
display(final_df)
