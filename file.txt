from pyspark.sql import functions as F

overall_summary_df = result_df.agg(
    F.count(
        F.when(F.col("transaction_count") > 0, 1)
    ).alias("cards_with_transactions"),

    F.coalesce(
        F.sum("transaction_count"), F.lit(0)
    ).alias("total_transactions"),

    F.coalesce(
        F.sum("total_amount_approved"), F.lit(0)
    ).alias("total_amount_approved")
)
