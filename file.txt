import re
import pandas as pd

# Example: Lists.IsBlacklisted [High] (500)
pattern = re.compile(
    r"^\s*(.*?)\s*\[[^\]]*\]\s*\(\s*(-?\d+(?:\.\d+)?)\s*\)\s*$"
)

def extract_facts(value):
    facts = {}

    if pd.isna(value):
        return facts

    for item in str(value).split("|"):
        match = pattern.match(item)

        if match:
            fact_name = match.group(1).strip()
            fact_value = float(match.group(2))
            facts[fact_name] = fact_value

    return facts

# Create fact columns, preserving the original row order and index

    [extract_facts(value) for value in apps["facts_fired"]],
    index=apps.index
).fillna(0)

apps = pd.concat([apps, fact_columns], axis=1)
