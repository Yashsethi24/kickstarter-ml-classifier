if df["APPLICATION_NUMBER"].isna().any():
    raise ValueError("APPLICATION_NUMBER contains missing values.")

df["fraud"] = pd.to_numeric(df["fraud"], errors="raise")
df["total_current_balance"] = pd.to_numeric(
    df["total_current_balance"], errors="raise"
)

if not df["fraud"].isin([0, 1]).all():
    raise ValueError("fraud must contain only 0 or 1.")

# Verify that each application's repeated rows have the same balance.
balance_counts = (
    df.groupby("APPLICATION_NUMBER")["total_current_balance"]
      .nunique(dropna=False)
)

if balance_counts.gt(1).any():
    raise ValueError(
        "Some applications have different balances across rows. "
        "Resolve card-level balances or historical snapshots first."
    )

# Create the two columns BEFORE grouping.
df["positive_balance"] = (
    df["total_current_balance"].clip(lower=0).fillna(0)
)

def clean_fact(value):
    value = str(value).strip()
    value = re.sub(r"\s*\([-+]?\d+(?:\.\d+)?\)\s*$", "", value)
    value = re.sub(r"\s*\[[^\]]*\]\s*$", "", value)
    return value.strip()

df["fact_set"] = df["facts_fired"].fillna("").map(
    lambda value: {
        clean_fact(part)
        for part in str(value).split("|")
        if part.strip()
    }
)

# Count each application once, combining all its fired facts.
df = (
    df.groupby("APPLICATION_NUMBER", as_index=False)
      .agg(
          fraud=("fraud", "max"),
          positive_balance=("positive_balance", "first"),
          fact_set=("fact_set", lambda values: set().union(*values)),
      )
)

remaining = pd.Series(True, index=df.index)
results = []

for position, row in enumerate(selected.itertuples(index=False), start=1):
    fact_name = clean_fact(row.fact)

    matched = remaining & df["fact_set"].map(
        lambda facts: fact_name in facts
    )

    captured = df.loc[matched]
    fraud_mask = captured["fraud"].eq(1)

    applications = len(captured)
    fraud_applications = int(fraud_mask.sum())
    nonfraud_applications = applications - fraud_applications

    results.append({
        "order": position,
        "fact": row.fact,
        "original_FPR": row.FPR,
        "incremental_applications": applications,
        "incremental_fraud_applications": fraud_applications,
        "incremental_nonfraud_applications": nonfraud_applications,
        "incremental_FPR": (
            nonfraud_applications / fraud_applications
            if fraud_applications else np.nan
        ),
        "incremental_fraud_balance": captured.loc[
            fraud_mask, "positive_balance"
        ].sum(),
    })

    remaining.loc[matched] = False

incremental_performance = pd.DataFrame(results)
