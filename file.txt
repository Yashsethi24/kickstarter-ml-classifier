import pandas as pd
from pyspark.sql import functions as F, Window

# ------------------------------------------------------------
# 1. Select cards with utilization >= 80%.
# report already contains one row per card and CR_LIM from SLIM.
# ------------------------------------------------------------
high_cards = report.loc[
    report["utilization_pct"].ge(80) & report["CR_LIM"].gt(0),
    ["CARD_NUMBER", "OPEN_DT", "CR_LIM", "utilization_pct"]
].copy()

high_cards["CARD_NUMBER"] = (
    high_cards["CARD_NUMBER"].astype("string").str.strip()
)
high_cards["OPEN_DT"] = pd.to_datetime(
    high_cards["OPEN_DT"], errors="raise"
).dt.normalize()

if high_cards["CARD_NUMBER"].duplicated().any():
    raise ValueError("Expected one row per CARD_NUMBER in report.")

if high_cards["OPEN_DT"].isna().any():
    raise ValueError("Some selected cards have a missing OPEN_DT.")

high_keys = spark.createDataFrame(
    [
        (str(card), opened.date(), float(limit))
        for card, opened, limit in high_cards[
            ["CARD_NUMBER", "OPEN_DT", "CR_LIM"]
        ].itertuples(index=False, name=None)
    ],
    schema="CARD_NUMBER string, OPEN_DT date, CR_LIM double"
)

# ------------------------------------------------------------
# 2. Read all transaction types for these cards.
# Do not apply AMOUNT_APPROVED > 0 here: payment records
# may have a different amount sign or a missing amount.
# ------------------------------------------------------------
events = (
    spark.table("tsys.v_cfm_tsys_mbna_bef")
    .select(
        F.trim(F.col("CARD_NUMBER").cast("string"))
        .alias("CARD_NUMBER"),

        F.trim(F.col("MESSAGE_TYPE").cast("string"))
        .alias("MESSAGE_TYPE"),

        F.col("AMOUNT_APPROVED").cast("double")
        .alias("AMOUNT_APPROVED"),

        F.to_timestamp("Auth_DateTime").alias("AUTH_TS")
    )
    .join(high_keys, "CARD_NUMBER", "inner")
)

# Keep dated events on or after card issuance.
dated_events = events.filter(
    F.col("AUTH_TS").isNotNull()
    & (F.to_date("AUTH_TS") >= F.col("OPEN_DT"))
)

is_payment = F.col("MESSAGE_TYPE") == F.lit("0108")

# ------------------------------------------------------------
# 3. Cumulative spending by transaction timestamp.
# Combine equal timestamps so threshold timing is unambiguous.
# ------------------------------------------------------------
spend_by_ts = (
    dated_events
    .filter(
        (~is_payment)
        & (F.col("AMOUNT_APPROVED") > 0)
    )
    .groupBy("CARD_NUMBER", "OPEN_DT", "CR_LIM", "AUTH_TS")
    .agg(
        F.sum("AMOUNT_APPROVED").alias("spend_at_timestamp")
    )
)

spend_window = (
    Window.partitionBy("CARD_NUMBER")
    .orderBy("AUTH_TS")
    .rowsBetween(Window.unboundedPreceding, Window.currentRow)
)

spend_timeline = spend_by_ts.withColumn(
    "cumulative_spend",
    F.sum("spend_at_timestamp").over(spend_window)
)

spend_timing = (
    spend_timeline
    .groupBy("CARD_NUMBER")
    .agg(
        F.min("AUTH_TS").alias("first_spend_ts"),
        F.max("AUTH_TS").alias("last_spend_ts"),

        F.sum("spend_at_timestamp")
        .alias("total_spend_since_open"),

        F.min(
            F.when(
                F.col("cumulative_spend") >= 0.80 * F.col("CR_LIM"),
                F.col("AUTH_TS")
            )
        ).alias("reached_80pct_ts")
    )
)

# ------------------------------------------------------------
# 4. Identify payment records since issuance.
# ------------------------------------------------------------
payments = dated_events.filter(is_payment)

payment_summary = (
    payments.groupBy("CARD_NUMBER")
    .agg(
        F.count("*").alias("payment_record_count"),
        F.min("AUTH_TS").alias("first_payment_ts"),
        F.max("AUTH_TS").alias("last_payment_ts")
    )
)

# Payments strictly after the card first reached 80%.
payments_after_80 = (
    payments
    .join(
        spend_timing.select("CARD_NUMBER", "reached_80pct_ts"),
        "CARD_NUMBER",
        "inner"
    )
    .filter(F.col("AUTH_TS") > F.col("reached_80pct_ts"))
    .groupBy("CARD_NUMBER")
    .agg(
        F.count("*").alias("payment_records_after_80"),
        F.min("AUTH_TS").alias("first_payment_after_80_ts")
    )
)

# Surface records whose dates cannot be used.
date_quality = (
    events.groupBy("CARD_NUMBER")
    .agg(
        F.sum(
            F.when(F.col("AUTH_TS").isNull(), 1).otherwise(0)
        ).alias("records_missing_auth_ts")
    )
)

# ------------------------------------------------------------
# 5. One result row per selected card.
# ------------------------------------------------------------
timing_spark = (
    high_keys
    .join(spend_timing, "CARD_NUMBER", "left")
    .join(payment_summary, "CARD_NUMBER", "left")
    .join(payments_after_80, "CARD_NUMBER", "left")
    .join(date_quality, "CARD_NUMBER", "left")
    .fillna(
        0,
        subset=[
            "payment_record_count",
            "payment_records_after_80",
            "records_missing_auth_ts"
        ]
    )
    .withColumn(
        "has_payment",
        F.col("payment_record_count") > 0
    )
    .withColumn(
        "has_payment_after_80",
        F.col("payment_records_after_80") > 0
    )
    .withColumn(
        "days_open_to_80pct",
        F.datediff(F.to_date("reached_80pct_ts"), "OPEN_DT")
    )
    .withColumn(
        "days_first_spend_to_80pct",
        F.datediff(
            F.to_date("reached_80pct_ts"),
            F.to_date("first_spend_ts")
        )
    )
    .withColumn(
        "days_80pct_to_first_payment",
        F.datediff(
            F.to_date("first_payment_after_80_ts"),
            F.to_date("reached_80pct_ts")
        )
    )
    .withColumn(
        "spend_pct_since_open",
        100 * F.col("total_spend_since_open") / F.col("CR_LIM")
    )
)

utilization_timing = timing_spark.toPandas()

utilization_timing = utilization_timing.merge(
    high_cards[["CARD_NUMBER", "utilization_pct"]],
    on="CARD_NUMBER",
    how="left",
    validate="one_to_one"
)

# Group cards by how quickly they reached 80%.
utilization_timing["time_to_80_band"] = pd.cut(
    utilization_timing["days_open_to_80pct"],
    bins=[0, 8, 15, 31, 61, 91, float("inf")],
    labels=[
        "0–7 days",
        "8–14 days",
        "15–30 days",
        "31–60 days",
        "61–90 days",
        "91+ days"
    ],
    right=False
).astype("string").fillna("80% not reached in dated spending")

timing_summary = (
    utilization_timing
    .groupby("time_to_80_band", as_index=False, sort=False)
    .agg(
        cards=("CARD_NUMBER", "nunique"),
        cards_with_payment=("has_payment", "sum"),
        cards_with_payment_after_80=("has_payment_after_80", "sum")
    )
)

display(utilization_timing)
display(timing_summary)
