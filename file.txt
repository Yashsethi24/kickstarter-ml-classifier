# Remove hyphens from phone numbers; leave nulls blank
phone = (
    df["Guaranter Phone"]
    .astype("string")
    .str.replace("-", "", regex=False)
    .str.strip()
    .fillna("")
)

# Copy the same phone into all three destination columns
for column in [
    "Residential_Phone_Number",
    "Business_Telephone",
    "Mobile_Telephone",
]:
    output_df[column] = phone

# Email address in uppercase; leave nulls blank
output_df["Email_Address"] = (
    df["Guaranter PersonEmail"]
    .astype("string")
    .str.strip()
    .str.upper()
    .fillna("")
)

# Birth date as YYYYMMDD; leave nulls or invalid dates blank
output_df["Primary_Date_of_Birth"] = (
    pd.to_datetime(
        df["Guaranter Birthdate"],
        format="%m/%d/%Y",
        errors="coerce",
    )
    .dt.strftime("%Y%m%d")
    .fillna("")
)
