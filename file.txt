# POS values "10" and "81" indicate card not present.
claims_df["CARD_PRESENCE"] = "Card Present"

claims_df.loc[
    claims_df["POS"].astype("string").str.strip().isin(["10", "81"]),
    "CARD_PRESENCE"
] = "Card Not Present"

card_presence_summary = (
    claims_df.groupby(["CLAIM_YM", "CARD_PRESENCE"])[CARD]
    .nunique()
    .unstack(fill_value=0)
    .reindex(columns=["Card Present", "Card Not Present"], fill_value=0)
    .rename(columns={
        "Card Present": "card_present_cards",
        "Card Not Present": "card_not_present_cards",
    })
    .sort_index()
    .reset_index()
)

display(card_presence_summary)
