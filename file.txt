import pandas as pd

df = apps.copy()
df["score"] = pd.to_numeric(df["score"], errors="coerce")
df["is_fraud"] = pd.to_numeric(
    df["Fraud_ind"], errors="coerce"
).eq(1).astype(int)

distribution = (
    df.groupby("score", dropna=False)
    .agg(
        total_applications=("is_fraud", "size"),
        fraud_count=("is_fraud", "sum")
    )
    .reset_index()
    .sort_values("score", ascending=False, na_position="last")
    .reset_index(drop=True)
)

total_apps = distribution["total_applications"].sum()
total_frauds = distribution["fraud_count"].sum()

distribution["fraud_rate_pct"] = (
    distribution["fraud_count"]
    / distribution["total_applications"] * 100
)

distribution["fraud_pct"] = (
    distribution["fraud_count"] / (total_frauds or 1) * 100
)

distribution["cumulative_applications"] = (
    distribution["total_applications"].cumsum()
)
distribution["cumulative_applications_pct"] = (
    distribution["cumulative_applications"] / (total_apps or 1) * 100
)
distribution["cumulative_frauds"] = (
    distribution["fraud_count"].cumsum()
)
distribution["cumulative_fraud_pct"] = (
    distribution["cumulative_frauds"] / (total_frauds or 1) * 100
)
distribution["cumulative_fraud_rate_pct"] = (
    distribution["cumulative_frauds"]
    / distribution["cumulative_applications"] * 100
)

pct_cols = distribution.columns[
    distribution.columns.str.endswith("_pct")
]
distribution[pct_cols] = distribution[pct_cols].round(2)
