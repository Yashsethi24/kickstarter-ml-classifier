%sql
CREATE OR REPLACE TEMP VIEW input_card_cifs AS
SELECT DISTINCT
    ac.acct_no AS input_card,
    cuac.cust_id,
    c.cust_no AS cif_id
FROM caedw_ca.account ac
INNER JOIN caedw_ca.customer_account cuac
    ON ac.acct_id = cuac.acct_id
INNER JOIN caedw_ca.customer c
    ON cuac.cust_id = c.cust_id
WHERE ac.acct_no IN (
    SELECT ACCT_NUM_TA1
    FROM hive_metastore.tsys.v_cfm_tsys_tdct_acct_slim
    WHERE TYPEC_VIP = 'A'
      AND ACCT_NUM_KEEP_TO = 0
);

%sql
CREATE OR REPLACE TEMP VIEW VIP_CIFS AS
SELECT DISTINCT
    '004CF0' || cif_id || 'C' AS name,
    'NULL' AS value
FROM input_card_cifs
WHERE cif_id NOT LIKE '%-%';
