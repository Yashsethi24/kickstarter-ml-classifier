def extract_facts(value):
    if pd.isna(value):
        return []

    # Keep only the fact name before "[".
    # Count each fact once per application.
    return list(dict.fromkeys(
        part.split("[", 1)[0].strip()
        for part in str(value).split("|")

    ))
