# Completed days between opening and the first transaction.
claims_df["FIRST_TRANS_DAYS_FROM_OPEN"] = (
    claims_df["FIRST_TRANS_HOURS_FROM_OPEN"] // 24
)

day_labels = [
    "0–1 days",
    "2–3 days",
    "4–10 days",
    "11–30 days",
    ">30 days",
]

claims_df["FIRST_TRANS_DAY_BUCKET"] = pd.cut(
    claims_df["FIRST_TRANS_DAYS_FROM_OPEN"],
    bins=[-1, 1, 3, 10, 30, float("inf")],
    labels=day_labels,
)

first_transaction_distribution = (
    claims_df.groupby(
        ["FIRST_TRANS_DAY_BUCKET", "CLAIM_YM"],
        observed=True,
    )[CARD]
    .nunique()
    .unstack(fill_value=0)
    .reindex(day_labels, fill_value=0)
    .sort_index(axis=1)
)

display(first_transaction_distribution)
