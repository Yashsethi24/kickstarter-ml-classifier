# 2. COMMON SQL QUERY
# ==========================================================

def get_base_query(report_type):

    if report_type == "monthly":

        report_start = "TRUNC(report_date, 'MM')"

        benchmark_start = """
            ADD_MONTHS(TRUNC(report_date, 'MM'), -6)
        """

    else:

        report_start = "DATE_SUB(report_date, 7)"

        benchmark_start = "DATE_SUB(report_date, 97)"

    return f"""

    WITH report_parameters AS (
        SELECT
            DATE_SUB(CURRENT_DATE(), 1) AS report_date
    ),

    date_parameters AS (
        SELECT
            report_date,

            {report_start} AS report_start,

            {benchmark_start} AS benchmark_start

        FROM report_parameters
    ),

    parameters AS (
        SELECT
            *,

            DATE_SUB(report_start, 1)
                AS benchmark_end,

            DATEDIFF(report_date, report_start) + 1
                AS report_days,

            DATEDIFF(report_start, benchmark_start)
                AS benchmark_days,

            DATEDIFF(LAST_DAY(report_date), report_date)
                AS remaining_days

        FROM date_parameters
    ),

    base_data AS (
        SELECT

            TO_DATE(Application_Date)
                AS application_date,

            CASE

                WHEN UPPER(TRIM(Address_Province))
                    IN ('ON', 'ONTARIO')
                    THEN 'ON'

                WHEN UPPER(TRIM(Address_Province))
                    IN ('QC', 'PQ', 'QUEBEC', 'QUÉBEC')
                    THEN 'QC'

                WHEN UPPER(TRIM(Address_Province))
                    IN ('AB', 'ALBERTA')
                    THEN 'AB'

                WHEN UPPER(TRIM(Address_Province))
                    IN ('BC', 'BRITISH COLUMBIA')
                    THEN 'BC'

                ELSE 'Others'

            END AS province_group,

            CASE

                WHEN Transit_Number = 'AMAZXX'
                    THEN 'AMAZON'

                ELSE 'MBNA'

            END AS channel,

            CAST(Application_Number AS STRING)
                AS application_number,

            COALESCE(
                TRY_CAST(
                    REGEXP_REPLACE(
                        CAST(Credit_Limit AS STRING),
                        ',',
                        ''
                    ) AS DECIMAL(18, 2)
                ),
                0
            ) AS credit_limit

        FROM hive_metastore.fiqm.applications

        CROSS JOIN parameters p

        WHERE Product_Type = 'PUCC'
            AND DATA_SOURCE = 'MBNA'
            AND Application_Number IS NOT NULL
            AND Application_Number RLIKE '^[A-E]'

            AND TO_DATE(Application_Date) BETWEEN
                p.benchmark_start AND p.report_date
    ),

    claims_data AS (
        SELECT

            CAST(Application_Number AS STRING)
                AS application_number,

            MAX(
                CASE
                    WHEN priority_alert_ind = 1
                        THEN 1
                    ELSE 0
                END
            ) AS priority_alert_ind,

            MAX(
                CASE
                    WHEN CONF_APP_FRAUD = 1
                        OR claim_ind = 1
                        THEN 1
                    ELSE 0
                END
            ) AS fraud_ind

        FROM claims_final_ss11

        GROUP BY CAST(Application_Number AS STRING)
    ),

    om5_data AS (
        SELECT DISTINCT

            CAST(Application_Number AS STRING)
                AS application_number

        FROM om5_alerts
    ),

    priority_data AS (
        SELECT

            CAST(Application_Number AS STRING)
                AS application_number,

            MAX(
                CASE
                    WHEN autoblock_alert_ind = 1
                        THEN 1
                    ELSE 0
                END
            ) AS autoblock_ind

        FROM priority_table

        GROUP BY CAST(Application_Number AS STRING)
    ),

    base_w_alerts AS (
        SELECT

            a.*,

            CASE
                WHEN b.priority_alert_ind = 1
                    OR c.application_number IS NOT NULL
                    THEN 1
                ELSE 0
            END AS alert_ind,

            COALESCE(b.fraud_ind, 0)
                AS fraud_ind,

            COALESCE(d.autoblock_ind, 0)
                AS autoblock_ind

        FROM base_data a

        LEFT JOIN claims_data b
            ON a.application_number = b.application_number

        LEFT JOIN om5_data c
            ON a.application_number = c.application_number

        LEFT JOIN priority_data d
            ON a.application_number = d.application_number
    ),

    channel_data AS (
        SELECT

            *,

            CASE
                WHEN alert_ind = 1
                    AND autoblock_ind = 1
                    THEN 1
                ELSE 0
            END AS autoblock_alert_ind,

            CASE
                WHEN alert_ind = 1
                    AND autoblock_ind = 0
                    THEN 1
                ELSE 0
            END AS non_autoblock_alert_ind

        FROM base_w_alerts

        WHERE channel IN ('MBNA', 'AMAZON')
    ),

    channel_list AS (
        SELECT channel

        FROM VALUES
            ('MBNA'),
            ('AMAZON')

        AS channel_list(channel)
    )
    """


# ==========================================================
# 3. FUNCTION TO GENERATE THRESHOLDS
# ==========================================================

def add_thresholds(df, report_type, prefix=""):

    if report_type == "monthly":

        green_limit = 125
        yellow_limit = 150

    else:

        green_limit = 110
        yellow_limit = 120

    for metric, threshold_name in threshold_metrics.items():

        actual = pd.to_numeric(
            df[f"{prefix}actual_{metric}"]
        )

        expected = pd.to_numeric(
            df[f"{prefix}expected_{metric}"]
        )

        if report_type == "monthly":

            yellow_condition = (
                actual * 100 <= expected * yellow_limit
            )

        else:

            yellow_condition = (
                actual * 100 < expected * yellow_limit
            )

        df[threshold_name] = np.select(

            [
                (expected == 0) & (actual == 0),

                (expected == 0) & (actual > 0),

                actual * 100 <= expected * green_limit,

                yellow_condition
            ],

            [
                "🟢Green",
                "🔴Red",
                "🟢Green",
                "🟠Yellow"
            ],

            default="🔴Red"
        )

    return df

# 4. MONTHLY AND 8-DAY AGGREGATED REPORTS
# ==========================================================

def generate_aggregated_report(report_type):

    is_monthly = report_type == "monthly"

    metrics = (
        monthly_metrics
        if is_monthly
        else daily_metrics
    )

    prefix = "mtd_" if is_monthly else ""

    aggregation_sql = ",\n".join(

        f"{metric_expressions[metric]} AS {metric}"

        for metric in metrics
    )

    metric_columns = []

    for metric in metrics:

        actual = f"COALESCE(a.{metric}, 0)"

        expected = f"""
            COALESCE(b.{metric}, 0)
            / NULLIF(p.benchmark_days, 0)
            * p.report_days
        """

        remaining = f"""
            COALESCE(b.{metric}, 0)
            / NULLIF(p.benchmark_days, 0)
            * p.remaining_days
        """

        if metric == "credit_limit":

            metric_columns.extend([

                f"""
                ROUND({actual}, 0)
                    AS {prefix}actual_{metric}
                """,

                f"""
                ROUND({expected}, 0)
                    AS {prefix}expected_{metric}
                """
            ])

            if is_monthly:

                metric_columns.append(f"""
                    ROUND({remaining}, 0)
                        AS remaining_{metric}
                """)

        else:

            metric_columns.extend([

                f"""
                CAST({actual} AS BIGINT)
                    AS {prefix}actual_{metric}
                """,

                f"""
                CAST(ROUND({expected}, 0) AS BIGINT)
                    AS {prefix}expected_{metric}
                """
            ])

            if is_monthly:

                metric_columns.append(f"""
                    CAST(ROUND({remaining}, 0) AS BIGINT)
                        AS remaining_{metric}
                """)

    metric_columns_sql = ",\n".join(metric_columns)

    query = get_base_query(report_type) + f"""

    ,

    period_data AS (
        SELECT

            b.*,

            CASE
                WHEN b.application_date >= p.report_start
                    THEN 'ACTUAL'
                ELSE 'BENCHMARK'
            END AS period_name

        FROM channel_data b

        CROSS JOIN parameters p
    ),

    aggregated_data AS (
        SELECT

            period_name,

            channel,

            COALESCE(province_group, 'Total')
                AS province_group,

            {aggregation_sql}

        FROM period_data

        GROUP BY GROUPING SETS (

            (period_name, channel, province_group),

            (period_name, channel)

        )
    ),

    benchmark AS (
        SELECT *

        FROM aggregated_data

        WHERE period_name = 'BENCHMARK'
    ),

    actual AS (
        SELECT *

        FROM aggregated_data

        WHERE period_name = 'ACTUAL'
    ),

    province_list AS (
        SELECT province_group

        FROM VALUES
            ('ON'),
            ('QC'),
            ('AB'),
            ('BC'),
            ('Others'),
            ('Total')

        AS province_list(province_group)
    ),

    report_groups AS (
        SELECT

            c.channel,

            g.province_group

        FROM channel_list c

        CROSS JOIN province_list g
    ),

    report_metrics AS (
        SELECT

            p.report_date,
            p.report_start,
            p.benchmark_start,
            p.benchmark_end,

            rg.channel,
            rg.province_group,

            {metric_columns_sql}

        FROM report_groups rg

        CROSS JOIN parameters p

        LEFT JOIN benchmark b
            ON rg.channel = b.channel
            AND rg.province_group = b.province_group

        LEFT JOIN actual a
            ON rg.channel = a.channel
            AND rg.province_group = a.province_group
    )

    SELECT *

    FROM report_metrics

    ORDER BY

        CASE channel
            WHEN 'MBNA' THEN 1
            WHEN 'AMAZON' THEN 2
            ELSE 3
        END,

        CASE province_group
            WHEN 'ON' THEN 1
            WHEN 'QC' THEN 2
            WHEN 'AB' THEN 3
            WHEN 'BC' THEN 4
            WHEN 'Others' THEN 5
            WHEN 'Total' THEN 6
            ELSE 7
        END

    """

    report = spark.sql(query).toPandas()

    report = add_thresholds(
        report,
        report_type,
        prefix
    )

    return report

# 5. DATE-WISE OVERALL 8-DAY REPORT
# ==========================================================

def generate_daily_overall_report():

    aggregation_sql = ",\n".join(

        f"{metric_expressions[metric]} AS {metric}"

        for metric in daily_metrics
    )

    complete_metrics_sql = ",\n".join(

        f"COALESCE(m.{metric}, 0) AS {metric}"

        for metric in daily_metrics
    )

    benchmark_metrics_sql = ",\n".join(

        f"AVG({metric}) AS expected_{metric}"

        for metric in daily_metrics
    )

    final_metric_columns = []

    for metric in daily_metrics:

        if metric == "credit_limit":

            final_metric_columns.extend([

                f"""
                ROUND(a.{metric}, 0)
                    AS actual_{metric}
                """,

                f"""
                ROUND(b.expected_{metric}, 0)
                    AS expected_{metric}
                """
            ])

        else:

            final_metric_columns.extend([

                f"""
                CAST(a.{metric} AS BIGINT)
                    AS actual_{metric}
                """,

                f"""
                CAST(
                    ROUND(b.expected_{metric}, 0)
                    AS BIGINT
                ) AS expected_{metric}
                """
            ])

    final_metric_columns_sql = ",\n".join(
        final_metric_columns
    )

    query = get_base_query("daily") + f"""

    ,

    daily_aggregation AS (
        SELECT

            application_date,

            channel,

            {aggregation_sql}

        FROM channel_data

        GROUP BY
            application_date,
            channel
    ),

    calendar_dates AS (
        SELECT

            EXPLODE(
                SEQUENCE(
                    benchmark_start,
                    report_date,
                    INTERVAL 1 DAY
                )
            ) AS application_date

        FROM parameters
    ),

    date_channel_grid AS (
        SELECT

            d.application_date,

            c.channel

        FROM calendar_dates d

        CROSS JOIN channel_list c
    ),

    complete_daily_metrics AS (
        SELECT

            g.application_date,

            DATE_FORMAT(g.application_date, 'EEEE')
                AS weekday,

            DAYOFWEEK(g.application_date)
                AS weekday_number,

            g.channel,

            {complete_metrics_sql}

        FROM date_channel_grid g

        LEFT JOIN daily_aggregation m
            ON g.application_date = m.application_date
            AND g.channel = m.channel
    ),

    benchmark AS (
        SELECT

            d.channel,

            d.weekday_number,

            COUNT(*) AS benchmark_weekday_days,

            {benchmark_metrics_sql}

        FROM complete_daily_metrics d

        CROSS JOIN parameters p

        WHERE d.application_date BETWEEN
            p.benchmark_start AND p.benchmark_end

        GROUP BY

            d.channel,

            d.weekday_number
    ),

    actual AS (
        SELECT d.*

        FROM complete_daily_metrics d

        CROSS JOIN parameters p

        WHERE d.application_date BETWEEN
            p.report_start AND p.report_date
    )

    SELECT

        p.report_date,

        p.report_start,

        p.benchmark_start,

        p.benchmark_end,

        a.application_date,

        a.weekday,

        a.channel,

        b.benchmark_weekday_days,

        {final_metric_columns_sql}

    FROM actual a

    CROSS JOIN parameters p

    LEFT JOIN benchmark b
        ON a.channel = b.channel
        AND a.weekday_number = b.weekday_number

    ORDER BY

        CASE a.channel
            WHEN 'MBNA' THEN 1
            WHEN 'AMAZON' THEN 2
            ELSE 3
        END,

        a.application_date DESC

    """

    report = spark.sql(query).toPandas()

    report = add_thresholds(
        report,
        "daily"
    )

    return report

# 7. ADD PERCENTAGE DIFFERENCES
# ==========================================================

def add_percentage_difference(df, metrics, prefix=""):

    for metric in metrics:

        actual = pd.to_numeric(
            df[f"{prefix}actual_{metric}"]
        )

        expected = pd.to_numeric(
            df[f"{prefix}expected_{metric}"]
        )

        percentage = (
            (actual - expected)
            .div(expected.replace(0, np.nan))
            .mul(100)
        )

        percentage = percentage.mask(
            (actual == 0) & (expected == 0),
            0
        )

        df[f"perc_diff_{metric}"] = percentage.round(1)

    return df

def plot_metric_comparison(

    df,

    actual_col,

    expected_col,

    title,

    y_label,

    save_filename,

    actual_label="Actual",

    expected_label="Expected",

    graph_type="province",

    channel=None,

    show_plot=True

):

    plot_df = df.copy()

    plot_df["channel"] = (
        plot_df["channel"]
        .astype(str)
        .str.strip()
        .str.upper()
    )

    if graph_type == "province":

        plot_df["province_group"] = (
            plot_df["province_group"]
            .astype(str)
            .str.strip()
            .str.upper()
        )

        plot_df["channel_sort"] = (
            plot_df["channel"]
            .map(channel_sort_order)
        )

        plot_df["province_sort"] = (
            plot_df["province_group"]
            .map(province_sort_order)
        )

        plot_df = plot_df.sort_values(
            ["channel_sort", "province_sort"]
        ).reset_index(drop=True)

        x_labels = (
            plot_df["province_group"]
            .replace({"OTHERS": "Others"})
            .tolist()
        )

        x_label = "Province"

    else:

        if channel is not None:

            plot_df = plot_df[
                plot_df["channel"] == channel
            ].copy()

        plot_df = (
            plot_df
            .sort_values("application_date")
            .reset_index(drop=True)
        )

        x_labels = (
            pd.to_datetime(plot_df["application_date"])
            .dt.strftime("%b %d\n%a")
            .tolist()
        )

        x_label = "Application Date"


    actual_values = (
        pd.to_numeric(
            plot_df[actual_col],
            errors="coerce"
        )
        .fillna(0)
        .to_numpy(dtype=float)
    )

    expected_values = (
        pd.to_numeric(
            plot_df[expected_col],
            errors="coerce"
        )
        .fillna(0)
        .to_numpy(dtype=float)
    )

    x = np.arange(len(plot_df))

    bar_width = 0.30

    fig, ax = plt.subplots(
        figsize=(15, 7),
        facecolor="#F2F2F2"
    )

    ax.set_facecolor("#FCFCFC")

    actual_bars = ax.bar(

        x - bar_width / 2,

        actual_values,

        width=bar_width,

        label=actual_label,

        color="#548235",

        edgecolor="grey",

        linewidth=1.2
    )

    expected_bars = ax.bar(

        x + bar_width / 2,

        expected_values,

        width=bar_width,

        label=expected_label,

        color="#A9D18E",

        edgecolor="grey",

        linewidth=1.2
    )

    ax.set_title(
        title,
        fontsize=12
    )

    ax.set_xlabel(
        x_label,
        fontsize=10,
        labelpad=8
    )

    ax.set_ylabel(
        y_label,
        fontsize=10,
        labelpad=8
    )

    ax.set_xticks(x)

    ax.set_xticklabels(
        x_labels,
        fontsize=9
    )

    ax.yaxis.set_major_locator(
        MaxNLocator(integer=True)
    )

    ax.yaxis.set_major_formatter(

        FuncFormatter(
            lambda value, position: f"{value:,.0f}"
        )
    )

    ax.tick_params(
        axis="y",
        labelsize=9
    )

    max_value = max(

        actual_values.max()
            if len(actual_values) else 0,

        expected_values.max()
            if len(expected_values) else 0,

        1
    )

    ax.set_ylim(
        0,
        max_value * 1.22
    )

    # Add MBNA and AMAZON headings
    # for province-wise graphs

    if graph_type == "province":

        mbna_count = (
            plot_df["channel"] == "MBNA"
        ).sum()

        amazon_count = (
            plot_df["channel"] == "AMAZON"
        ).sum()

        separator_position = mbna_count - 0.5

        if mbna_count > 0 and amazon_count > 0:

            ax.axvline(

                x=separator_position,

                color="#7A7A7A",

                linestyle="--",

                linewidth=1.2
            )

        mbna_center = (
            mbna_count - 1
        ) / 2

        amazon_center = (
            mbna_count
            + (amazon_count - 1) / 2
        )

        if mbna_count > 0:

            ax.text(

                mbna_center,

                max_value * 1.14,

                "MBNA",

                ha="center",

                fontsize=12,

                fontweight="bold"
            )

        if amazon_count > 0:

            ax.text(

                amazon_center,

                max_value * 1.14,

                "AMAZON",

                ha="center",

                fontsize=12,

                fontweight="bold"
            )

    ax.legend(
        loc="upper right"
    )

    ax.grid(

        axis="y",

        linestyle="--",

        linewidth=0.7,

        alpha=0.25
    )

    ax.set_axisbelow(True)

    ax.spines["top"].set_visible(False)

    ax.spines["right"].set_visible(False)

    ax.spines["left"].set_alpha(0.4)

    ax.spines["bottom"].set_alpha(0.4)

    actual_labels = ax.bar_label(

        actual_bars,

        labels=[
            f"{value:,.0f}"
            for value in actual_values
        ],

        padding=3,

        fontsize=10
    )

    expected_labels = ax.bar_label(

        expected_bars,

        labels=[
            f"{value:,.0f}"
            for value in expected_values
        ],

        padding=3,

        fontsize=10
    )

    for label in actual_labels:

        xp, yp = label.get_position()

        label.set_position(
            (xp - 2, yp)
        )

    for label in expected_labels:

        xp, yp = label.get_position()

        label.set_position(

# 16. FUNCTION TO FORMAT AGGREGATED REPORT TABLES
# ==========================================================

def format_summary_table(

    df,

    report_type,

    include_province=True

):

    is_monthly = report_type == "monthly"

    prefix = "mtd_" if is_monthly else ""

    metrics = (
        monthly_metrics
        if is_monthly
        else daily_metrics
    )

    columns = ["channel"]

    rename_columns = {
        "channel": "Channel"
    }

    if include_province:

        columns.append("province_group")

        rename_columns["province_group"] = "Province"

    for metric, label in metrics.items():

        actual_col = f"{prefix}actual_{metric}"

        expected_col = f"{prefix}expected_{metric}"

        columns.extend([
            actual_col,
            expected_col
        ])

        rename_columns[actual_col] = (
            f"Actual {label}"
        )

        rename_columns[expected_col] = (
            f"Expected {label}"
        )

        if is_monthly:

            remaining_col = f"remaining_{metric}"

            columns.append(remaining_col)

            rename_columns[remaining_col] = (
                f"Remaining {label}"
            )

        if metric in threshold_metrics:

            threshold_col = threshold_metrics[metric]

            columns.append(threshold_col)

            rename_columns[threshold_col] = (
                f"{label} Status"
            )

    table = df[columns].copy()

    for col in table.columns:

        if "credit_limit" in col:

            table[col] = table[col].apply(

                lambda value: (
                    f"${float(value):,.0f}"
                    if pd.notna(value)
                    else "N/A"
                )
            )

        elif (

            col.startswith("mtd_actual_")

            or col.startswith("mtd_expected_")

            or col.startswith("actual_")

            or col.startswith("expected_")

            or col.startswith("remaining_")

        ):

            table[col] = table[col].apply(

                lambda value: (
                    f"{float(value):,.0f}"
                    if pd.notna(value)
                    else "N/A"
                )
            )

    table.rename(
        columns=rename_columns,
        inplace=True
    )

    return table

# 17. FUNCTION TO CREATE DAILY DATE-WISE TABLES
# ==========================================================

def create_daily_metric_table(

    df,

    channel,

    metric

):

    metric_labels = {

        "application_count": (
            "Apps",
            "Avg_Apps"
        ),

        "credit_limit": (
            "Credit_Limit",
            "Avg_Credit_Limit"
        ),

        "alerts_count": (
            "Alerts",
            "Avg_Alerts"
        ),

        "frauds_count": (
            "Frauds",
            "Avg_Frauds"
        ),

        "autoblock_alerts_count": (
            "Autoblock",
            "Avg_Autoblock"
        ),

        "non_autoblock_alerts_count": (
            "Non_Autoblock",
            "Avg_Non_Autoblock"
        )
    }

    actual_label, expected_label = (
        metric_labels[metric]
    )

    columns = [

        "application_date",

        "weekday",

        "channel",

        f"actual_{metric}",

        f"expected_{metric}",

        f"perc_diff_{metric}"
    ]

    threshold_col = threshold_metrics.get(metric)

    if threshold_col:

        columns.append(threshold_col)

    table = (

        df[df["channel"] == channel]

        .sort_values(
            "application_date",
            ascending=False
        )

        [columns]

        .copy()
    )

    rename_columns = {

        "application_date": "APPLICATION_DATE",

        "weekday": "WEEKDAY",

        "channel": "Product",

        f"actual_{metric}": actual_label,

        f"expected_{metric}": expected_label,

        f"perc_diff_{metric}": "Perc_Diff"
    }

    if threshold_col:

        rename_columns[threshold_col] = "STATUS"

    table.rename(

        columns=rename_columns,

        inplace=True
    )

    table["APPLICATION_DATE"] = (

        pd.to_datetime(
            table["APPLICATION_DATE"]
        )

        .dt.strftime("%Y-%m-%d")
    )

    table["Perc_Diff"] = (

        table["Perc_Diff"].apply(

            lambda value: (

                f"{value:.1f}%"

                if pd.notna(value)

                else "N/A"
            )
        )
    )

    for col in [

        actual_label,

        expected_label
    ]:

        if metric == "credit_limit":

            table[col] = table[col].apply(

                lambda value: f"${float(value):,.0f}"

                if pd.notna(value)

                else "N/A"
            )

        else:

            table[col] = table[col].apply(

                lambda value: f"{float(value):,.0f}"

                if pd.notna(value)

                else "N/A"
            )

    return table.reset_index(drop=True)

def dataframe_to_html(df):

        df.to_html(

            index=False,

                font-family:Arial, Helvetica, sans-serif;
                font-size:11px;
                color:#222222;
                border:1px solid #BFBFBF;
            ">
            """
        )

        .replace(

            '<tr style="text-align: right;">',

            '<tr style="text-align:center;">'
        )

        .replace(

            "<th>",

            """
            <th style="
                background-color:#548235;
                color:white;
                border:1px solid #BFBFBF;
                padding:6px;
                text-align:center;
            ">
            """
        )

        .replace(

            "<td>",

            """
            <td style="
                border:1px solid #BFBFBF;
                padding:5px;
                text-align:center;
            ">
            """
        )
    )
