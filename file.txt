from itertools import combinations
import numpy as np
import pandas as pd

def rank_fact_combinations(
    df,
    fact_cols,
    fraud_col="fraud_flag",
    combination_size=2,
    min_applications=100,
):
    data = df.loc[df[fraud_col].notna()].copy()

    if data.empty:
        raise ValueError("No applications with a known fraud label.")

    if not data[fraud_col].isin([0, 1]).all():
        raise ValueError("Fraud marker must contain 0 or 1.")

    scores = data[fact_cols].apply(pd.to_numeric, errors="raise")
    present = scores.fillna(0).ne(0)
    fraud = data[fraud_col].eq(1)

    total_fraud = fraud.sum()
    baseline_rate = fraud.mean()

    # A combination cannot have more hits than any individual member
    eligible_facts = present.columns[
        present.sum() >= min_applications
    ]

    rows = []

    for facts in combinations(eligible_facts, combination_size):
        hit = present[list(facts)].all(axis=1)

        n_present = int(hit.sum())
        if n_present < min_applications:
            continue

        fraud_present = int((hit & fraud).sum())
        n_absent = int((~hit).sum())
        fraud_absent = int((~hit & fraud).sum())

        rate_present = fraud_present / n_present
        rate_absent = fraud_absent / n_absent if n_absent else np.nan

        # Compare against the strongest individual fact in this group
        best_single_rate = max(
            fraud[present[fact]].mean() for fact in facts
        )

        rows.append({
            "facts": " + ".join(facts),
            "applications_present": n_present,
            "fraud_present": fraud_present,
            "fraud_rate_present": rate_present,
            "fraud_rate_absent": rate_absent,
            "lift": (
                rate_present / baseline_rate
                if baseline_rate else np.nan
            ),
            "fraud_capture": (
                fraud_present / total_fraud
                if total_fraud else np.nan
            ),
            "application_coverage": n_present / len(data),
            "best_single_fact_fraud_rate": best_single_rate,
            "gain_over_best_single_pp": (
                rate_present - best_single_rate
            ) * 100,
        })

    result = pd.DataFrame(rows)

    if result.empty:
        return result

    return result.sort_values(
        ["lift", "fraud_present"], ascending=False
    ).reset_index(drop=True)

pairs = rank_fact_combinations(
    df,
    fact_cols,
    fraud_col="fraud_flag",
    combination_size=2,
    min_applications=100,  # Adjust for your population size
)
