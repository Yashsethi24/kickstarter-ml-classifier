# Read credit limits from MBNA SLIM
mbna_limits = spark.sql("""
    SELECT ACCT_NUM_TA1, CR_LIM
    FROM tsys.v_cfm_tsys_mbna_acct_slim
    WHERE ACCT_NUM_TA1 IS NOT NULL
""").toPandas()

# Align card numbers as strings, preserving leading zeros
base["ACCT_NUM_TA1"] = (
    base["ACCT_NUM_TA1"].astype("string").str.strip().replace("", pd.NA)
)
mbna_limits["ACCT_NUM_TA1"] = (
    mbna_limits["ACCT_NUM_TA1"]
    .astype("string").str.strip().replace("", pd.NA)
)

# Remove identical key/limit pairs
mbna_limits = (
    mbna_limits.dropna(subset=["ACCT_NUM_TA1"])
    .drop_duplicates(subset=["ACCT_NUM_TA1", "CR_LIM"])
)

# Add credit limit without multiplying base rows
base = base.merge(
    mbna_limits,
    on="ACCT_NUM_TA1",
    how="left",
    validate="many_to_one"
)
