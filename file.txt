from pathlib import Path
import pandas as pd

BASE_PATH = Path(
    "/Workspace/Users/rajshreyas.penukonda@td.com/"
    "Applications Fraud/TU IDX/Backbook"
)

JOIN_KEY = "unique_application_id"  # Replace with the exact column name
MAPPING_FILE = "application_id_mapping.csv"

categories = {
    "Genuines": "genuine",
    "Frauds": "fraud",
    "DLQ": "DLQ",
}

dataframes = []

for portfolio in ["TDCT", "MBNA"]:
    for folder_name, category in categories.items():
        folder = BASE_PATH / portfolio / folder_name
        mapping_path = folder / MAPPING_FILE

        # Find the main CSV, excluding the mapping file.
        main_files = [
            path for path in folder.iterdir()
            if path.suffix.lower() == ".csv"
            and path.name.lower() != MAPPING_FILE.lower()
        ]

        if len(main_files) != 1:
            raise ValueError(
                f"Expected exactly one main CSV in {folder}; "
                f"found {len(main_files)}."
            )

        # Read as strings to preserve leading zeros in application IDs.
        main_df = pd.read_csv(main_files[0], dtype=str)
        mapping_df = pd.read_csv(mapping_path, dtype=str)

        for name, df in [("main", main_df), ("mapping", mapping_df)]:
            if JOIN_KEY not in df.columns:
                raise ValueError(
                    f"{JOIN_KEY!r} missing from {name} CSV in {folder}. "
                    f"Available columns: {df.columns.tolist()}"
                )

        # Exclude missing mapping keys so pandas cannot match null IDs.
        mapping_df = mapping_df.loc[
            mapping_df[JOIN_KEY].notna()
            & mapping_df[JOIN_KEY].ne("")
        ]

        joined_df = main_df.merge(
            mapping_df,
            on=JOIN_KEY,
            how="left",
            suffixes=("", "_mapping"),
            validate="many_to_one",
        )

        joined_df["category"] = category
        joined_df["portfolio"] = portfolio

        dataframes.append(joined_df)

final_df = pd.concat(dataframes, ignore_index=True, sort=False)

display(final_df)
