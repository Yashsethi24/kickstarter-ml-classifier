%sql
WITH params AS (
    SELECT
        CAST('${start_date}' AS DATE) AS start_date,
        CAST('${end_date}' AS DATE) AS end_date
),

application_base AS (
    SELECT
        DATE_FORMAT(a.APPLICATION_DATE, 'yyyy-MM-01') AS month,
        CASE
            WHEN a.DATA_SOURCE = 'MBNA'
                 OR a.APPLICATION_NUMBER RLIKE '^[A-E]'
            THEN CASE
                     WHEN a.TRANSIT_NUMBER = 'AMAZXX' THEN 'Amazon'
                     ELSE 'MBNA'
                 END
            ELSE CASE
                     WHEN a.TRANSIT_NUMBER = '03157' THEN 'TDCT Online'
                     ELSE 'TDCT Branch'
                 END
        END AS channel,
        a.APPLICATION_NUMBER
    FROM hive_metastore.fiqm.applications a
    CROSS JOIN params p
    WHERE a.APPLICATION_DATE BETWEEN p.start_date AND p.end_date
      AND (
          a.DATA_SOURCE = 'MBNA'
          OR a.APPLICATION_NUMBER RLIKE '^[A-E]'
          OR (
              a.PRODUCT_TYPE = 'PUCC'
              AND a.APPLICATION_NUMBER NOT RLIKE '^[A-E]'
              AND a.DECISION_PATH <> 'SAMD'
          )
      )
),

applications AS (
    SELECT
        month,
        channel,
        COUNT(DISTINCT APPLICATION_NUMBER) AS application_cnt
    FROM application_base
    GROUP BY month, channel
),

account_base AS (
    SELECT
        'MBNA' AS portfolio,
        CASE WHEN BRANCH = 'AMAZXX' THEN 'Amazon'
             ELSE 'MBNA'
        END AS channel,
        ACCT_NUM_TA1,
        CAST(OPEN_DT AS DATE) AS open_date,
        CAST(SEC_FRAUD_STAT_DT AS DATE) AS secure_fraud_date,
        STATC_TIERED_AUTH,
        STC_SECURE_FRAUD,
        CUR_BAL_TOT
    FROM tsys.v_cfm_tsys_mbna_acct_slim

    UNION ALL

    SELECT
        'TDCT' AS portfolio,
        CASE WHEN BRANCH IN ('3157', '03157') THEN 'TDCT Online'
             ELSE 'TDCT Branch'
        END AS channel,
        ACCT_NUM_TA1,
        CAST(OPEN_DT AS DATE) AS open_date,
        CAST(SEC_FRAUD_STAT_DT AS DATE) AS secure_fraud_date,
        STATC_TIERED_AUTH,
        STC_SECURE_FRAUD,
        CUR_BAL_TOT
    FROM tsys.v_cfm_tsys_tdct_acct_slim
),

account_flags AS (
    SELECT
        DATE_FORMAT(s.open_date, 'yyyy-MM-01') AS month,
        s.channel,
        s.ACCT_NUM_TA1,
        s.CUR_BAL_TOT,

        CASE
            WHEN s.portfolio = 'MBNA'
                 AND s.STATC_TIERED_AUTH IN ('L7', 'L8')
                THEN 1
            WHEN s.portfolio = 'TDCT'
                 AND s.STATC_TIERED_AUTH = 'L7'
                THEN 1
            ELSE 0
        END AS suspected_flag,

        CASE
            WHEN s.portfolio = 'MBNA'
                 AND s.STATC_TIERED_AUTH IN ('L7', 'L8')
                 AND COALESCE(s.STC_SECURE_FRAUD, '') = ''
                THEN 1
            WHEN s.portfolio = 'TDCT'
                 AND s.STATC_TIERED_AUTH = 'L7'
                 AND COALESCE(s.STC_SECURE_FRAUD, '') <> 'FA'
                THEN 1
            ELSE 0
        END AS suspected_no_overlap_flag,

        CASE
            WHEN s.portfolio = 'MBNA'
                 AND s.STC_SECURE_FRAUD != ''
                 AND DATEDIFF(
                     s.secure_fraud_date, s.open_date
                 ) <= 90
                THEN 1
            WHEN s.portfolio = 'TDCT'
                 AND s.STC_SECURE_FRAUD = 'FA'
                THEN 1
            ELSE 0
        END AS confirmed_flag

    FROM account_base s
    CROSS JOIN params p
    WHERE s.open_date BETWEEN p.start_date AND p.end_date
),

fraud AS (
    SELECT
        month,
        channel,
        COUNT(DISTINCT ACCT_NUM_TA1) AS open_cnt,

        COUNT(DISTINCT CASE
            WHEN suspected_flag = 1 THEN ACCT_NUM_TA1
        END) AS suspected_fraud_cnt,

        COUNT(DISTINCT CASE
            WHEN suspected_no_overlap_flag = 1 THEN ACCT_NUM_TA1
        END) AS suspected_fraud_cnt_no_overlap,

        COUNT(DISTINCT CASE
            WHEN confirmed_flag = 1 THEN ACCT_NUM_TA1
        END) AS confirmed_fraud_cnt,

        SUM(CASE
            WHEN suspected_no_overlap_flag = 1
                THEN COALESCE(CUR_BAL_TOT, 0)
            ELSE 0
        END) AS suspected_fraud_amt,

        SUM(CASE
            WHEN confirmed_flag = 1
                THEN COALESCE(CUR_BAL_TOT, 0)
            ELSE 0
        END) AS confirmed_fraud_amt

    FROM account_flags
    GROUP BY month, channel
)

SELECT
    a.month AS Month,
    a.channel AS Channel,
    a.application_cnt AS Application_Count,
    COALESCE(f.open_cnt, 0) AS Open_Count,
    COALESCE(f.suspected_fraud_cnt_no_overlap, 0) AS Suspected_Fraud,
    COALESCE(f.confirmed_fraud_cnt, 0) AS Confirmed_Fraud_Count,
    COALESCE(f.suspected_fraud_amt, 0) AS Suspected_Fraud_Amount,
    COALESCE(f.confirmed_fraud_amt, 0) AS Confirmed_Fraud_Amount,

    1.0 * COALESCE(f.confirmed_fraud_cnt, 0)
        / NULLIF(a.application_cnt, 0) AS Confirmed_Fraud_Rate,

    -- Preserves the original rate, including overlap.
    1.0 * COALESCE(f.suspected_fraud_cnt, 0)
        / NULLIF(a.application_cnt, 0) AS Suspected_Fraud_Rate

FROM applications a
LEFT JOIN fraud f
    ON a.month = f.month
   AND a.channel = f.channel
ORDER BY Channel, Month;
