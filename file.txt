# Step 1: Filter applications where any decline fact was fired
filtered_df = final_df[
    final_df['FACTS_FIRED'].fillna('').apply(
        lambda x: any(fact in x for fact in decline_facts)
    )
].copy()

# Step 2: Monthly summary
monthly_summary = (
    filtered_df
    .groupby('APPLICATION_MON')
    .agg(
        Total_Applications=('APPLICATION_NUMBER', 'nunique'),

        Non_Fraud_Applications=(
            'APPLICATION_NUMBER',
            lambda x: x[filtered_df.loc[x.index, 'FRAUD'] == 0].nunique()
        ),

        Fraud_Applications=(
            'APPLICATION_NUMBER',
            lambda x: x[filtered_df.loc[x.index, 'FRAUD'] == 1].nunique()
        ),

        Fraud_Current_Balance=(
            'TOTAL_CURRENT_BALANCE',
            lambda x: x[filtered_df.loc[x.index, 'FRAUD'] == 1].sum()
        )
    )
    .reset_index()
)

display(monthly_summary)
