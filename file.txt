%sql
WITH ais AS (
    SELECT DISTINCT TRIM(CAST(customerAcctNumber AS STRING)) AS acct
    FROM cad_fico.ais
    WHERE customerAcctNumber IS NOT NULL
),
account AS (
    SELECT DISTINCT TRIM(CAST(header_customerAcctNumber AS STRING)) AS acct
    FROM cad_fico.account
    WHERE header_customerAcctNumber IS NOT NULL
)

SELECT 'AIS Total' AS metric, COUNT(*) AS cnt FROM ais

UNION ALL

SELECT 'Account Total', COUNT(*) FROM account

UNION ALL

SELECT 'Common Accounts', COUNT(*)
FROM ais
INNER JOIN account USING (acct)

UNION ALL

SELECT 'Only in AIS', COUNT(*)
FROM ais
LEFT ANTI JOIN account USING (acct)

UNION ALL

SELECT 'Only in Account', COUNT(*)
FROM account
LEFT ANTI JOIN ais USING (acct);



%sql
SELECT DISTINCT a.customerAcctNumber
FROM cad_fico.ais a
LEFT ANTI JOIN cad_fico.account b
    ON TRIM(CAST(a.customerAcctNumber AS STRING)) =
       TRIM(CAST(b.header_customerAcctNumber AS STRING));



%sql
SELECT DISTINCT b.header_customerAcctNumber
FROM cad_fico.account b
LEFT ANTI JOIN cad_fico.ais a
    ON TRIM(CAST(b.header_customerAcctNumber AS STRING)) =
       TRIM(CAST(a.customerAcctNumber AS STRING));
