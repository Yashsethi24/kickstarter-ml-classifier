WITH daily_data AS (
    SELECT
        DATE(load_date) AS app_date,
        COUNT(DISTINCT application_number) AS num_apps,
        COUNT(DISTINCT CASE WHEN priority_alert_ind_final = 1
            THEN application_number END) AS num_total_alerts,
        COUNT(DISTINCT CASE WHEN autoblock_alert_ind = 1
            THEN application_number END) AS num_AB_alerts,
        COUNT(DISTINCT CASE WHEN priority_alert_ind_final = 1
            AND autoblock_alert_ind = 0
            THEN application_number END) AS num_score_alerts,
        COUNT(DISTINCT CASE WHEN CONFIRMED_FRAUD = 1
            THEN application_number END) AS num_CG_Frauds,
        COUNT(DISTINCT CASE WHEN SUSPECTED_FRAUD_IND = 1
            THEN application_number END) AS num_Susp_Frauds
    FROM aml_qf_apps
    WHERE DATE(load_date) BETWEEN '2026-10-01' AND '2026-11-22'
    GROUP BY DATE(load_date)
),

oct_avg AS (
    SELECT
        AVG(num_apps) AS avg_apps,
        AVG(num_total_alerts) AS avg_total_alerts,
        AVG(num_AB_alerts) AS avg_AB_alerts,
        AVG(num_score_alerts) AS avg_score_alerts,
        AVG(num_CG_Frauds) AS avg_CG_Frauds,
        AVG(num_Susp_Frauds) AS avg_Susp_Frauds
    FROM daily_data
    WHERE app_date BETWEEN '2026-10-01' AND '2026-10-31'
      AND app_date NOT IN ('2026-10-07', '2026-10-08')
),

adjusted_data AS (
    SELECT
        d.app_date,

        CASE WHEN d.app_date IN ('2026-10-07', '2026-10-08')
             THEN a.avg_apps ELSE d.num_apps END AS num_apps,

        CASE WHEN d.app_date IN ('2026-10-07', '2026-10-08')
             THEN a.avg_total_alerts ELSE d.num_total_alerts
        END AS num_total_alerts,

        CASE WHEN d.app_date IN ('2026-10-07', '2026-10-08')
             THEN a.avg_AB_alerts ELSE d.num_AB_alerts
        END AS num_AB_alerts,

        CASE WHEN d.app_date IN ('2026-10-07', '2026-10-08')
             THEN a.avg_score_alerts ELSE d.num_score_alerts
        END AS num_score_alerts,

        CASE WHEN d.app_date IN ('2026-10-07', '2026-10-08')
             THEN a.avg_CG_Frauds ELSE d.num_CG_Frauds
        END AS num_CG_Frauds,

        CASE WHEN d.app_date IN ('2026-10-07', '2026-10-08')
             THEN a.avg_Susp_Frauds ELSE d.num_Susp_Frauds
        END AS num_Susp_Frauds

    FROM daily_data d
    CROSS JOIN oct_avg a
)

SELECT
    DATE_TRUNC('MONTH', app_date) AS app_mon,
    ROUND(SUM(num_apps), 0) AS num_apps,
    ROUND(SUM(num_total_alerts), 0) AS num_total_alerts,
    ROUND(SUM(num_AB_alerts), 0) AS num_AB_alerts,
    ROUND(SUM(num_score_alerts), 0) AS num_score_alerts,
    ROUND(SUM(num_CG_Frauds), 0) AS num_CG_Frauds,
    ROUND(SUM(num_Susp_Frauds), 0) AS num_Susp_Frauds
FROM adjusted_data
GROUP BY DATE_TRUNC('MONTH', app_date)
ORDER BY app_mon;
