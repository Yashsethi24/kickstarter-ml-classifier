overall_summary = (
    ever_over_80
    .filter(F.col("AUTH_month") != "2026-10")
    .agg(
        F.count("*").alias("unique_cards_ever_over_80"),
        F.count(
            F.when(F.col("has_later_payment") == 1, 1)
        ).alias("with_later_payment"),
        F.count(
            F.when(F.col("has_later_payment") == 0, 1)
        ).alias("without_observed_later_payment")
    )
)
