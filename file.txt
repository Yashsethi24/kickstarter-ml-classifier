import pandas as pd
import networkx as nx

def remove_cross_duplicates(df):

    id_col = 'Unique_record_ID'
    app_col = 'Application_Number'

    # Preserve original data and index
    temp = df.reset_index(drop=True).copy()

    # Remove exact duplicate ID-application pairs
    pairs = temp[[id_col, app_col]].dropna().drop_duplicates()

    # Create graph
    G = nx.Graph()

    for uid, app in pairs.itertuples(index=False, name=None):
        G.add_edge(('ID', uid), ('APP', app))

    # Find maximum one-to-one mapping
    matching = nx.algorithms.bipartite.maximum_matching(
        G,
        top_nodes={('ID', x) for x in pairs[id_col].unique()}
    )

    # Get selected ID-application pairs
    selected_pairs = {
        (node[1], partner[1])
        for node, partner in matching.items()
        if node[0] == 'ID'
    }

    # Keep only selected combinations
    mask = [
        (uid, app) in selected_pairs
        for uid, app in zip(temp[id_col], temp[app_col])
    ]

    final_df = (
        temp.loc[mask]
        .drop_duplicates(subset=[id_col, app_col])
        .reset_index(drop=True)
    )

    return final_df


final_df = remove_cross_duplicates(df)
