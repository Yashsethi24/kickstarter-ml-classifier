import os
import base64
from io import BytesIO

import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
from matplotlib.ticker import FuncFormatter

plt.rcParams["font.family"] = "DejaVu Sans"

# ----------------------------------------------------------
# 1. Clean the existing channel and province columns
# ----------------------------------------------------------
mbna_report["channel"] = (
    mbna_report["channel"]
    .astype(str)
    .str.strip()
    .str.upper()
)

mbna_report["province_group"] = (
    mbna_report["province_group"]
    .astype(str)
    .str.strip()
    .str.upper()
    .replace({
        "ONTARIO": "ON",
        "QUEBEC": "QC",
        "QUÉBEC": "QC",
        "PQ": "QC",
        "ALBERTA": "AB",
        "BRITISH COLUMBIA": "BC",
        "OTHER": "OTHERS"
    })
)

# ----------------------------------------------------------
# 2. Create temporary sort columns
# ----------------------------------------------------------
channel_sort_order = {
    "MBNA": 1,
    "AMAZON": 2
}

province_sort_order = {
    "ON": 1,
    "QC": 2,
    "AB": 3,
    "BC": 4,
    "OTHERS": 5
}

mbna_report["channel_sort"] = mbna_report["channel"].map(channel_sort_order)
mbna_report["province_sort"] = mbna_report["province_group"].map(province_sort_order)

mbna_report.sort_values(
    ["channel_sort", "province_sort"],
    inplace=True
)

mbna_report.reset_index(drop=True, inplace=True)

# ----------------------------------------------------------
# 3. Folder where graphs will be saved
#    Change this path if you want another location
# ----------------------------------------------------------
output_dir = "mbna_graphs"
os.makedirs(output_dir, exist_ok=True)

# ----------------------------------------------------------
# 4. Reusable plotting function
# ----------------------------------------------------------
def plot_metric_comparison(
    df,
    actual_col,
    expected_col,
    title,
    y_label,
    save_filename,
    actual_label="MTD Actual",
    expected_label="MTD Expected",
    show_plot=True
):
    province_labels = (
        df["province_group"]
        .replace({"OTHERS": "Others"})
        .tolist()
    )

    actual_values = (
        pd.to_numeric(df[actual_col], errors="coerce")
        .fillna(0)
        .to_numpy()
    )

    expected_values = (
        pd.to_numeric(df[expected_col], errors="coerce")
        .fillna(0)
        .to_numpy()
    )

    x = np.arange(len(df))
    bar_width = 0.30

    # Find where MBNA ends and AMAZON begins
    mbna_count = (df["channel"] == "MBNA").sum()
    separator_position = mbna_count - 0.5

    # Create the graph
    fig, ax = plt.subplots(
        figsize=(15, 7),
        facecolor="#F2F2F2"
    )

    ax.set_facecolor("#FCFCFC")

    actual_bars = ax.bar(
        x - bar_width / 2,
        actual_values,
        width=bar_width,
        label=actual_label,
        color="#548235",
        edgecolor="grey",
        linewidth=1.2
    )

    expected_bars = ax.bar(
        x + bar_width / 2,
        expected_values,
        width=bar_width,
        label=expected_label,
        color="#A9D18E",
        edgecolor="grey",
        linewidth=1.2
    )

    # Title and axis labels
    ax.set_title(
        title,
        fontsize=12
    )

    ax.set_xlabel(
        "Province",
        fontsize=10,
        labelpad=8
    )

    ax.set_ylabel(
        y_label,
        fontsize=10,
        labelpad=8
    )

    ax.set_xticks(x)
    ax.set_xticklabels(
        province_labels,
        fontsize=9
    )

    ax.yaxis.set_major_formatter(
        FuncFormatter(lambda value, position: f"{value:,.0f}")
    )

    ax.tick_params(
        axis="y",
        labelsize=9
    )

    # Add separator between MBNA and AMAZON
    ax.axvline(
        x=separator_position,
        color="#7A7A7A",
        linestyle="--",
        linewidth=1.2
    )

    # Add channel headings
    max_value = max(
        actual_values.max() if len(actual_values) > 0 else 0,
        expected_values.max() if len(expected_values) > 0 else 0
    )

    upper_limit = max_value * 1.22 if max_value > 0 else 1
    ax.set_ylim(0, upper_limit)

    mbna_center = (mbna_count - 1) / 2 if mbna_count > 0 else 0
    amazon_count = len(df) - mbna_count
    amazon_center = mbna_count + (amazon_count - 1) / 2 if amazon_count > 0 else mbna_count

    if mbna_count > 0:
        ax.text(
            mbna_center,
            max_value * 1.14 if max_value > 0 else 0.5,
            "MBNA",
            ha="center",
            fontsize=12,
            fontweight="bold"
        )

    if amazon_count > 0:
        ax.text(
            amazon_center,
            max_value * 1.14 if max_value > 0 else 0.5,
            "AMAZON",
            ha="center",
            fontsize=12,
            fontweight="bold"
        )

    # Legend, grid and borders
    ax.legend()
    ax.grid(axis="y", alpha=0.25)

    ax.grid(
        axis="y",
        linestyle="--",
        linewidth=0.7,
        alpha=0.25
    )

    ax.set_axisbelow(True)
    ax.spines["top"].set_visible(False)
    ax.spines["right"].set_visible(False)
    ax.spines["left"].set_alpha(0.4)
    ax.spines["bottom"].set_alpha(0.4)

    # Add bar labels
    actual_labels = ax.bar_label(
        actual_bars,
        labels=[f"{value:,.0f}" for value in actual_values],
        padding=3,
        fontsize=10,
        fontweight="normal"
    )

    expected_labels = ax.bar_label(
        expected_bars,
        labels=[f"{value:,.0f}" for value in expected_values],
        padding=3,
        fontsize=10,
        fontweight="normal"
    )

    # Move labels slightly apart
    for label in actual_labels:
        x_position, y_position = label.get_position()
        label.set_position((x_position - 2, y_position))

    for label in expected_labels:
        x_position, y_position = label.get_position()
        label.set_position((x_position + 2, y_position))

    fig.tight_layout()

    # ------------------------------------------------------
    # Save graph as PNG file
    # ------------------------------------------------------
    save_path = os.path.join(output_dir, save_filename)
    fig.savefig(
        save_path,
        format="png",
        dpi=150,
        bbox_inches="tight"
    )

    # ------------------------------------------------------
    # Save graph into memory and encode as base64
    # ------------------------------------------------------
    buffer = BytesIO()
    fig.savefig(
        buffer,
        format="png",
        dpi=150,
        bbox_inches="tight"
    )
    buffer.seek(0)

    graph_base64 = base64.b64encode(
        buffer.read()
    ).decode("utf-8")

    buffer.close()

    if show_plot:
        plt.show()

    plt.close(fig)

    # ------------------------------------------------------
    # HTML block for email body
    # ------------------------------------------------------
    graph_html = f"""
    <h3>{title}</h3>
    <img
        src="data:image/png;base64,{graph_base64}"
        width="900"
        style="max-width:100%; height:auto;"
    >
    """

    return {
        "save_path": save_path,
        "base64": graph_base64,
        "html": graph_html
    }

# ----------------------------------------------------------
# 5. Create and save Application graph
# ----------------------------------------------------------
application_graph_info = plot_metric_comparison(
    df=mbna_report,
    actual_col="mtd_actual_application_count",
    expected_col="mtd_expected_application_count",
    title="MBNA Application Volume: MTD Actual vs Expected",
    y_label="Application Count",
    save_filename="mbna_application_count.png",
    show_plot=True
)

# ----------------------------------------------------------
# 6. Create and save Alert graph
# ----------------------------------------------------------
alert_graph_info = plot_metric_comparison(
    df=mbna_report,
    actual_col="mtd_actual_alerts_count",
    expected_col="mtd_expected_alerts_count",
    title="MBNA Alert Volume: MTD Actual vs Expected",

# ----------------------------------------------------------

# ----------------------------------------------------------
fraud_graph_info = plot_metric_comparison(

    expected_col="mtd_expected_frauds_count",
    title="MBNA Fraud Volume: MTD Actual vs Expected",

# ----------------------------------------------------------
# 8. File paths of saved images
# ----------------------------------------------------------
application_graph_path = application_graph_info["save_path"]

print("Saved graph files:")

print(fraud_graph_path)

# ----------------------------------------------------------
# 9. Base64 strings (if needed later)
# ----------------------------------------------------------
mbna_application_graph = application_graph_info["base64"]
mbna_alert_graph = alert_graph_info["base64"]

# ----------------------------------------------------------
# 10. Individual HTML blocks (if needed later)

mbna_application_html = application_graph_info["html"]

# ----------------------------------------------------------
# 11. Combined HTML block for email body
# ----------------------------------------------------------
mbna_graphs_html = f"""

<br><br>

"""

# ----------------------------------------------------------
# 12. Remove temp columns
# ----------------------------------------------------------
mbna_report.drop(
    columns=["channel_sort", "province_sort"],
    inplace=True
)
