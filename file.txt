import pandas as pd

df = final_df.copy()

# Normalize dates so the difference uses calendar days.
open_dt = pd.to_datetime(df["OPEN_DT"], errors="coerce").dt.normalize()
claim_dt = pd.to_datetime(df["CLM_DT"], errors="coerce").dt.normalize()

df["days_to_claim"] = (claim_dt - open_dt).dt.days

labels = [
    "0-3 days",
    "4-10 days",
    "11-15 days",
    "16-30 days",
    "31-60 days",
    "61-90 days",
    "91-180 days",
    "181+ days",
]

df["claim_day_bucket"] = pd.cut(
    df["days_to_claim"],
    bins=[-1, 3, 10, 15, 30, 60, 90, 180, float("inf")],
    labels=labels,
    right=True,
)

summary_df = (
    df.groupby(["CLM_YM", "claim_day_bucket"], observed=False)
      .size()
      .unstack("claim_day_bucket", fill_value=0)
      .reindex(columns=labels, fill_value=0)
      .rename_axis(columns=None)
      .reset_index()
)

summary_df["total"] = summary_df[labels].sum(axis=1)

display(summary_df)
