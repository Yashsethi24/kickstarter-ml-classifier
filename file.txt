import pandas as pd

frauds = apps.loc[
    pd.to_numeric(apps["Fraud_ind"], errors="coerce").eq(1)
].copy()

frauds["score"] = pd.to_numeric(frauds["score"], errors="coerce")

fraud_distribution = (
    frauds.groupby("score", dropna=False)
    .size()
    .reset_index(name="fraud_count")
    .sort_values("score", ascending=False, na_position="last")
    .reset_index(drop=True)
)

total_frauds = fraud_distribution["fraud_count"].sum()

fraud_distribution["fraud_pct"] = (
    fraud_distribution["fraud_count"] / (total_frauds or 1) * 100
)

fraud_distribution["cumulative_frauds"] = (
    fraud_distribution["fraud_count"].cumsum()
)

fraud_distribution["cumulative_fraud_pct"] = (
    fraud_distribution["cumulative_frauds"] / (total_frauds or 1) * 100
)

fraud_distribution[["fraud_pct", "cumulative_fraud_pct"]] = (
    fraud_distribution[["fraud_pct", "cumulative_fraud_pct"]].round(2)
)

display(fraud_distribution)
