from pyspark.sql import functions as F

# 1. Multiply new weights by 100 and truncate decimals.
# Examples: 0.456 -> 45; -0.456 -> -45
integer_weights = weights[["fact", "weight"]].copy()
integer_weights["weight_int"] = (
    integer_weights["weight"] * 100
).astype("int64")

weights_spark = spark.createDataFrame(
    [
        (str(fact), int(weight))
        for fact, weight in integer_weights[
            ["fact", "weight_int"]
        ].itertuples(index=False, name=None)
    ],
    schema="fact string, weight_int long"
)

# 2. Count each fired fact once per application.
fired_facts = long_df.select(ID, "fact").distinct()

# Ensure every fired fact has a new weight.
unmatched = fired_facts.join(weights_spark, on="fact", how="left_anti")

if unmatched.limit(1).count():
    unmatched.select("fact").distinct().show(truncate=False)
    raise ValueError("Some fired facts have no corresponding new weight.")

# 3. Sum NEW integer weights, without multiplying by original fact scores.
application_totals = (
    fired_facts
    .join(weights_spark, on="fact", how="inner")
    .groupBy(ID)
    .agg(F.sum("weight_int").alias("total_score"))
)

# Assumes one row per application in model_df.
# Applications with no fired facts receive a score of zero.
application_scores = (
    model_df.select(ID, TARGET)
    .join(application_totals, on=ID, how="left")
    .fillna(0, subset=["total_score"])
)

# 4. Create buckets: 0–9, 10–19, 20–29, etc.
bucketed = (
    application_scores
    .withColumn(
        "bucket_start",
        (F.floor(F.col("total_score") / F.lit(10)) * 10).cast("long")
    )
    .withColumn("bucket_end", F.col("bucket_start") + 9)
)

# 5. Count genuine and fraud applications in each populated bucket.
score_distribution = (
    bucketed
    .groupBy("bucket_start", "bucket_end")
    .agg(
        F.count("*").alias("total_applications"),
        F.sum(
            F.when(F.col(TARGET) == 0, 1).otherwise(0)
        ).alias("genuine_applications"),
        F.sum(
            F.when(F.col(TARGET) == 1, 1).otherwise(0)
        ).alias("fraud_applications")
    )
    .withColumn(
        "fraud_percentage",
        F.round(
            100.0 * F.col("fraud_applications")
            / F.col("total_applications"),
            2
        )
    )
    .orderBy("bucket_start")
)

display(application_scores)
