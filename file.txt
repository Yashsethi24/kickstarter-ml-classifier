import pandas as pd

# Convert score to numeric
fact_df['score'] = pd.to_numeric(fact_df['score'], errors='coerce')

# Create score groups
fact_df['score_group'] = pd.cut(
    fact_df['score'],
    bins=[-float('inf'), 190, 200, float('inf')],
    labels=['0-190', '191-200', '200+']
)

# Count unique applications per group
summary = (
    fact_df.groupby('score_group', observed=True)['application_id']
    .nunique()
    .reset_index(name='unique_applications')
)

display(summary)
