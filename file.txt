    LEAST(
        NULLIF(TRY_CAST(App_FICO_FRAUD_Marked_Date AS DATE), DATE '0001-01-01'),
        NULLIF(TRY_CAST(FIQM_App_Fraud_Marked_Date AS DATE), DATE '0001-01-01'),

    )
) OVER (
    PARTITION BY header_applicationId
