
summary_df = (
    final_df.groupby('CLM_YM')
    .agg(
        unique_applications=('Application_Number', 'nunique'),
        applications_with_transactions=(
            'Application_Number',
            lambda x: x[final_df.loc[x.index, 'transaction_count'] > 0].nunique()
        ),
        total_transactions=('transaction_count', 'sum'),
        total_amount_approved=('total_amount_approved', 'sum')
    )
    .reset_index()
    .sort_values('CLM_YM')
)

display(summary_df)
