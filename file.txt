# Application-to-account mapping
application_accounts = (
    spark.table("hive_metastore.fiqm.applications")
    .join(application_ids, "APPLICATION_NUMBER", "left_semi")
    .select(
        "APPLICATION_NUMBER",
        F.trim("ACCOUNT_NUMBER").alias("ACCOUNT_NUMBER"),
    )
    .filter(
        F.col("ACCOUNT_NUMBER").isNotNull()
        & (F.col("ACCOUNT_NUMBER") != "")
    )
    .dropDuplicates(["APPLICATION_NUMBER", "ACCOUNT_NUMBER"])
    .withColumn(
        "join_key",
        F.expr("try_cast(ACCOUNT_NUMBER AS DECIMAL(38,0))"),
    )
)

# SLIM data with the corrected join key
slim_source = (
    spark.table("tsys.v_cfm_tsys_mbna_acct_slim")
    .withColumn(
        "join_key",
        F.expr("try_cast(APPLID AS DECIMAL(38,0))"),
    )
)

# Add application number and application-table account number to SLIM
slim = (
    slim_source.alias("s")
    .join(
        application_accounts.alias("a"),
        F.col("s.join_key") == F.col("a.join_key"),
        "inner",
    )
    .select(
        F.col("a.APPLICATION_NUMBER"),
        F.col("a.ACCOUNT_NUMBER").alias("APPLICATION_ACCOUNT_NUMBER"),
        *[
            F.col(f"s.{col}")
            for col in slim_source.columns
            if col != "join_key"
        ],
    )
)

display(slim)
