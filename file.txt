%sql
WITH combined AS (
    SELECT DISTINCT
        DATE_FORMAT(openDate, 'yyyy-MM') AS month,
        TRIM(CAST(Account_Number AS STRING)) AS acct,
        1 AS in_ais,
        0 AS in_account
    FROM final_data_ais_df
    WHERE Account_Number IS NOT NULL
      AND openDate >= '2026-01-01'
      AND openDate < '2026-10-01'

    UNION ALL

    SELECT DISTINCT
        DATE_FORMAT(openDate, 'yyyy-MM') AS month,
        TRIM(CAST(Account_Number AS STRING)) AS acct,
        0 AS in_ais,
        1 AS in_account
    FROM final_data_account_df
    WHERE Account_Number IS NOT NULL
      AND openDate >= '2026-01-01'
      AND openDate < '2026-10-01'
),

comparison AS (
    SELECT
        month,
        acct,
        MAX(in_ais) AS in_ais,
        MAX(in_account) AS in_account
    FROM combined
    GROUP BY month, acct
)

SELECT
    month,

    SUM(in_ais) AS ais_total,

    SUM(in_account) AS account_total,

    SUM(CASE
        WHEN in_ais = 1 AND in_account = 1
        THEN 1 ELSE 0
    END) AS common_accounts,

    SUM(CASE
        WHEN in_ais = 1 AND in_account = 0
        THEN 1 ELSE 0
    END) AS only_in_ais,

    SUM(CASE
        WHEN in_ais = 0 AND in_account = 1
        THEN 1 ELSE 0
    END) AS only_in_account

FROM comparison
GROUP BY month
ORDER BY month;
