from html import escape


def render_email_tree(tree):
    def render_node(item):
        children = item.get("children", [])

        box = f"""
        <table role="presentation" align="center"
               cellpadding="8" cellspacing="0" border="1"
               style="border-collapse:collapse;border:1px solid #b8c8d8;">
            <tr>
                <td align="center" bgcolor="#edf3f9">
                    <b>{escape(str(item["label"]))}</b><br>
                    {item["count"]:,} applications<br>
                    Loss: {item["amount"]:,.2f}
                </td>
            </tr>
        </table>
        """

        if not children:
            return box

        child_cells = "".join(
            f"""
            <td valign="top" align="center" style="padding:0 6px;">
                <div style="font-size:18px;color:#789;">&#8595;</div>
                {render_node(child)}
            </td>
            """
            for child in children
        )

        return f"""
        <table role="presentation" align="center"
               cellpadding="0" cellspacing="0" border="0">
            <tr>
                <td colspan="{len(children)}" align="center">
                    {box}
                </td>
            </tr>
            <tr>{child_cells}</tr>
        </table>
        """

    return render_node(tree)
