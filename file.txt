import pandas as pd
import networkx as nx

def remove_cross_duplicates(df):
    id_col = 'Unique_record_ID'
    app_col = 'Application_Number'

    # Remove exact duplicate mappings
    pairs = df[[id_col, app_col]].dropna().drop_duplicates()

    # Create bipartite graph
    G = nx.Graph()

    for uid, app in pairs.itertuples(index=False, name=None):
        G.add_edge(('ID', uid), ('APP', app))

    # Find maximum one-to-one matching
    matching = nx.algorithms.bipartite.maximum_matching(
        G,
        top_nodes={('ID', uid) for uid in pairs[id_col].unique()}
    )

    # Extract matched combinations
    matched_pairs = [
        (uid[1], app[1])
        for uid, app in matching.items()
        if uid[0] == 'ID'
    ]

    matched_df = pd.DataFrame(
        matched_pairs,
        columns=[id_col, app_col]
    )

    # Retain all original columns
    final_df = (
        df.merge(matched_df, on=[id_col, app_col], how='inner')
          .drop_duplicates(subset=[id_col, app_col])
          .reset_index(drop=True)
    )

    return final_df
