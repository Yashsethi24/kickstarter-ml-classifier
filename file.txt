import pandas as pd
from pyspark.sql import functions as F

# ------------------------------------------------------------
# One record per card for the card and transaction reports.
# Use latest OPEN_DT; stop if statuses or credit limits conflict.
# ------------------------------------------------------------
cards = base.dropna(subset=["CARD_NUMBER"]).copy()

cards["CR_LIM"] = pd.to_numeric(cards["CR_LIM"], errors="raise")

latest_run = cards.groupby("CARD_NUMBER")["OPEN_DT"].transform("max")
cards = cards.loc[cards["OPEN_DT"].eq(latest_run)].copy()

status_cols = ["SUSP_IND_BASE", "CNF_FRAUD_IND"]
check_cols = status_cols + ["CR_LIM"]

conflicts = (
    cards.groupby("CARD_NUMBER")[check_cols]
    .nunique(dropna=False)
)

if conflicts.gt(1).any(axis=None):
    raise ValueError(
        "Conflicting statuses or credit limits for a card "
        "at its latest OPEN_DT."
    )

cards = cards[
    ["CARD_NUMBER", "OPEN_DT", "CR_LIM"] + status_cols
].drop_duplicates()

cards["blocked"] = cards["SUSP_IND_BASE"].eq(1)

cards["unblocked"] = (
    cards["SUSP_IND_BASE"].eq(0)
    & cards["CNF_FRAUD_IND"].eq(0)
)

cards["confirmed_fraud"] = cards["CNF_FRAUD_IND"].eq(1)

# ------------------------------------------------------------
# Approved transactions, limited to cards in this report.
# Credit limit comes from base.CR_LIM.
# ------------------------------------------------------------
card_keys = spark.createDataFrame(
    cards[["CARD_NUMBER", "OPEN_DT"]].assign(
        OPEN_DT=lambda x: x["OPEN_DT"].dt.date
    )
)

tx = (
    spark.table("tsys.v_cfm_tsys_mbna_bef")
    .select(
        F.trim(
            F.col("CARD_NUMBER").cast("string")
        ).alias("CARD_NUMBER"),
        F.col("AMOUNT_APPROVED")
        .cast("double")
        .alias("AMOUNT_APPROVED"),
        F.to_timestamp("Auth_DateTime").alias("AUTH_TS")
    )
    .filter(F.col("AMOUNT_APPROVED") > 0)
    .join(card_keys, "CARD_NUMBER", "inner")
)

by_run = F.to_date("AUTH_TS") <= F.col("OPEN_DT")
after_run = F.to_date("AUTH_TS") > F.col("OPEN_DT")

tx_summary = (
    tx.groupBy("CARD_NUMBER")
    .agg(
        F.count("*").alias("transaction_count"),

        F.sum("AMOUNT_APPROVED")
        .alias("total_approved_amount"),

        F.sum(
            F.when(by_run, F.col("AMOUNT_APPROVED"))
            .otherwise(0.0)
        ).alias("approved_by_OPEN_DT"),

        F.sum(
            F.when(after_run, F.col("AMOUNT_APPROVED"))
            .otherwise(0.0)
        ).alias("approved_after_OPEN_DT"),

        F.sum(
            F.when(F.col("AUTH_TS").isNull(), 1)
            .otherwise(0)
        ).alias("transactions_missing_date")
    )
    .toPandas()
)

# ------------------------------------------------------------
# Combine card details, SLIM credit limits, and transactions.
# ------------------------------------------------------------
report = cards.merge(
    tx_summary,
    on="CARD_NUMBER",
    how="left",
    validate="one_to_one"
)

amount_count_cols = [
    "transaction_count",
    "total_approved_amount",
    "approved_by_OPEN_DT",
    "approved_after_OPEN_DT",
    "transactions_missing_date"
]

report[amount_count_cols] = report[amount_count_cols].fillna(0)

# Missing or nonpositive credit limits produce missing utilization.
report["utilization_pct"] = (
    report["total_approved_amount"]
    .div(report["CR_LIM"].where(report["CR_LIM"] > 0))
    .mul(100)
)

display(report)
