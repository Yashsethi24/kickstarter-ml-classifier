%sql
WITH bef_flags AS (
    SELECT
        CARD_NUMBER,
        MAX(
            CASE WHEN AMOUNT_APPROVED > 0 THEN 1 ELSE 0 END
        ) AS monetary_transaction_flag,
        MAX(
            CASE
                WHEN ACCT_STATUS1 LIKE 'PD%'
                  OR ACCT_STATUS2 LIKE 'PD%'
                  OR ACCT_STATUS3 LIKE 'PD%'
                  OR ACCT_STATUS4 LIKE 'PD%'
                THEN 1 ELSE 0
            END
        ) AS delinquency_index
    FROM tsys.v_cfm_tsys_mbna_bef
    GROUP BY CARD_NUMBER
)
SELECT
    s.ACCT_NUM AS card_number,
    CASE WHEN s.CRV_STAT_DT IS NOT NULL
         THEN 1 ELSE 0 END AS activation_flag,
    CASE WHEN s.STF_ACTV_YR = 'Y'
         THEN 1 ELSE 0 END AS activity_this_year_flag,
    s.OPEN_DT,
    CASE WHEN s.STC_CR_REVOKED IS NOT NULL
         THEN 1 ELSE 0 END AS credit_revoked_flag,
    CASE WHEN s.STF_FRAUD IS NOT NULL
         THEN 1 ELSE 0 END AS fraud_flag,
    CASE WHEN s.STATC_TIERED_AUTH LIKE 'L%'
         THEN 1 ELSE 0 END AS block_flag,
    COALESCE(b.monetary_transaction_flag, 0)
        AS monetary_transaction_flag,
    COALESCE(b.delinquency_index, 0)
        AS delinquency_index
FROM tsys.v_cfm_tsys_mbna_acct_slim s
LEFT JOIN bef_flags b
    ON s.ACCT_NUM = b.CARD_NUMBER;
