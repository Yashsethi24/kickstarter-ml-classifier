import pandas as pd

def validate_record_ids(df, id_col='Unique_record_ID'):

    ids = df[id_col].astype('string').str.strip()

    # Validate ID format: A + YYYYMMDD + 4 digits
    valid_format = ids.str.fullmatch(r'A\d{8}\d{4}').fillna(False)

    # Check invalid IDs
    invalid_ids = df.loc[~valid_format].copy()

    # Extract valid IDs
    valid_ids = ids[valid_format]

    temp = pd.DataFrame({
        'Record_ID': valid_ids,
        'Date': valid_ids.str[1:9],
        'Sequence': valid_ids.str[9:13].astype(int)
    })

    # Check duplicate IDs
    duplicate_ids = (
        temp[temp.duplicated('Record_ID', keep=False)]
        .sort_values('Record_ID')
    )

    # Check missing sequence numbers for each date
    missing_list = []
    summary_list = []

    for date, group in temp.groupby('Date'):

        sequences = set(group['Sequence'])
        max_seq = max(sequences)

        expected = set(range(1, max_seq + 1))
        missing = sorted(expected - sequences)

        duplicates = group['Record_ID'].duplicated().sum()

        for seq in missing:
            missing_list.append({
                'Date': date,
                'Missing_ID': f'A{date}{seq:04d}'
            })

        summary_list.append({
            'Date': date,
            'Total_Records': len(group),
            'Unique_IDs': group['Record_ID'].nunique(),
            'Expected_Records': max_seq,
            'Missing_Count': len(missing),
            'Duplicate_Count': duplicates
        })

    missing_df = pd.DataFrame(
        missing_list, columns=['Date', 'Missing_ID']
    )

    summary_df = pd.DataFrame(summary_list)

    return summary_df, missing_df, duplicate_ids, invalid_ids
