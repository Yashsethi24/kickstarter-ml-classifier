from datetime import datetime, timedelta
from dateutil.relativedelta import relativedelta

# First day of month, 3 months before MTD start date
start_date_3m = (
    datetime.strptime(mtd_start_date, "%Y-%m-%d")
    - relativedelta(months=3)
).strftime("%Y-%m-%d")

# Last Friday relative to end_date (inclusive if Friday)
last_friday = (
    datetime.strptime(end_date, "%Y-%m-%d")
    - timedelta(days=(datetime.strptime(end_date, "%Y-%m-%d").weekday() - 4) % 7)
).strftime("%Y-%m-%d")

print(start_date_3m, last_friday)


query = f"""
SELECT
    date_format(to_date(application_date, 'yyyyMMdd'), 'yyyyMM') AS year_mon,
    COUNT(DISTINCT Application_Number) AS num_apps
FROM final_data
WHERE application_date >= '{start_date_3m.replace("-", "")}'
  AND application_date <= '{last_friday.replace("-", "")}'
GROUP BY date_format(to_date(application_date, 'yyyyMMdd'), 'yyyyMM')
"""

display(spark.sql(query))
